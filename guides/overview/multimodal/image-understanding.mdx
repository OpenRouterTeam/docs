Requests with images, to multimodal models, are available via the [Responses API](/api_reference/responses/overview) (`/api/v1/responses`) with a multi-part message in the `input` parameter. Each image is an `input_image` content part whose `image_url` can either be a URL or a base64-encoded data URL. Note that multiple images can be sent in separate content array entries. The number of images you can send in a single request varies per provider and per model. Due to how the content is parsed, we recommend sending the text prompt first, then the images. If the images must come first, put them before the text within the same user message; system messages should contain text only.

OpenRouter supports both **direct URLs** and **base64-encoded data** for images:

- **URLs**: More efficient for publicly accessible images as they don't require local encoding
- **Base64**: Required for local files or private images that aren't publicly accessible

### Using Image URLs

Here's how to send an image using a URL:

<CodeGroup>

```typescript title="TypeScript SDK" expandable lines
import { OpenRouter } from '@openrouter/sdk';

const openRouter = new OpenRouter({
  apiKey: '<OPENROUTER_API_KEY>',
});

const result = await openRouter.responses.send({
  responsesRequest: {
    model: 'google/gemini-3-flash-preview',
    input: [
      {
        role: 'user',
        content: [
          {
            type: 'input_text',
            text: "What's in this image?",

          {
            type: 'input_image',

            detail: 'auto',
          },
        ],
      },
    ],
    stream: false,
  },
});

console.log(result);
```

```go title="Go" lines
package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	"log"
	"net/http"
	"os"
)

func main() {
	url := "https://openrouter.ai/api/v1/responses"

	inputItems := []map[string]any{
		{
			"role": "user",
			"content": []map[string]any{
				{
					"type": "input_text",
					"text": "What's in this image?",
				},
				{
					"type":      "input_image",
					"image_url": "https://upload.wikimedia.org/wikipedia/commons/a/a7/Camponotus_flavomarginatus_ant.jpg",
					"detail":    "auto",
				},
			},
		},
	}

	payload, _ := json.Marshal(map[string]any{
		"model": "google/gemini-3-flash-preview",
		"input": inputItems,
	})

	req, _ := http.NewRequest("POST", url, bytes.NewReader(payload))
	req.Header.Set("Authorization", "Bearer "+os.Getenv("OPENROUTER_API_KEY"))
	req.Header.Set("Content-Type", "application/json")

	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		log.Fatal(err)
	}
	defer resp.Body.Close()

	var result any
	if err := json.NewDecoder(resp.Body).Decode(&result); err != nil {
		log.Fatal(err)
	}
	out, _ := json.MarshalIndent(result, "", "  ")
	fmt.Println(string(out))
}
```

### Using Base64 Encoded Images

For locally stored images, you can send them using base64 encoding. Here's how to do it:

<CodeGroup>

```typescript title="TypeScript SDK" expandable lines
import { OpenRouter } from '@openrouter/sdk';
import * as fs from 'fs';

const openRouter = new OpenRouter({
  apiKey: '<OPENROUTER_API_KEY>',
});

async function encodeImageToBase64(imagePath: string): Promise<string> {
  const imageBuffer = await fs.promises.readFile(imagePath);
  const base64Image = imageBuffer.toString('base64');
  return `data:image/jpeg;base64,${base64Image}`;
}

// Read and encode the image
const imagePath = 'path/to/your/image.jpg';
const base64Image = await encodeImageToBase64(imagePath);

const result = await openRouter.responses.send({
  responsesRequest: {
    model: 'google/gemini-3-flash-preview',
    input: [
      {
        role: 'user',
        content: [
          {
            type: 'input_text',
            text: "What's in this image?",
          },
          {
            type: 'input_image',
            imageUrl: base64Image,
            detail: 'auto',
          },
        ],
      },
    ],
    stream: false,
  },
});

console.log(result);
```

```go title="Go" lines
package main

import (
	"bytes"
	"encoding/base64"
	"encoding/json"
	"fmt"
	"log"
	"net/http"
	"os"
)

func encodeImageToBase64(imagePath string) (string, error) {
	data, err := os.ReadFile(imagePath)
	if err != nil {
		return "", err
	}
	return "data:image/jpeg;base64," + base64.StdEncoding.EncodeToString(data), nil
}

func main() {
	// Read and encode the image
	imagePath := "path/to/your/image.jpg"
	base64Image, err := encodeImageToBase64(imagePath)
	if err != nil {
		log.Fatal(err)
	}

	payload, _ := json.Marshal(map[string]any{
		"model": "google/gemini-3-flash-preview",
		"input": []map[string]any{
			{
				"role": "user",
				"content": []map[string]any{
					{"type": "input_text", "text": "What's in this image?"},
					{"type": "input_image", "image_url": base64Image, "detail": "auto"},
				},
			},
		},
		"stream": false,
	})

	req, _ := http.NewRequest("POST", "https://openrouter.ai/api/v1/responses", bytes.NewReader(payload))
	req.Header.Set("Authorization", "Bearer "+os.Getenv("OPENROUTER_API_KEY"))
	req.Header.Set("Content-Type", "application/json")

	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		log.Fatal(err)
	}
	defer resp.Body.Close()

	var result any
	if err := json.NewDecoder(resp.Body).Decode(&result); err != nil {
		log.Fatal(err)
	}
	out, _ := json.MarshalIndent(result, "", "  ")
	fmt.Println(string(out))
}
```
