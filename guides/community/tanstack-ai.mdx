## TanStack AI

You can use [TanStack AI](https://tanstack.com/ai) to integrate OpenRouter with your React, Solid, or Preact applications. The OpenRouter adapter provides access to 400+ AI models from various providers through a single unified API. To get started, install [@tanstack/ai-openrouter](https://www.npmjs.com/package/@tanstack/ai-openrouter):

```bash lines
npm install @tanstack/ai @tanstack/ai-openrouter
```

### Basic Usage

<CodeGroup>

```typescript title="TypeScript" lines
import { chat } from "@tanstack/ai";
import { openRouterText } from "@tanstack/ai-openrouter";

const stream = chat({
  adapter: openRouterText("openai/gpt-5.2"),
  threadId: "conversation-1",
  messages: [{ role: "user", content: "Hello!" }],
});

for await (const chunk of stream) {
  if ("delta" in chunk && typeof chunk.delta === "string") {
    process.stdout.write(chunk.delta);
  }
}
console.log();

```

```python title="Python" lines
import os
from openrouter import OpenRouter

with OpenRouter(api_key=os.environ["OPENROUTER_API_KEY"]) as client:
    stream = client.chat.send(
        model="openai/gpt-5.2",
        messages=[{"role": "user", "content": "Hello!"}],
        stream=True,
    )

    for chunk in stream:
        if chunk.choices and chunk.choices[0].delta.content:
            print(chunk.choices[0].delta.content, end="", flush=True)
    print()

```

```go title="Go" lines
package main

import (
	"bufio"
	"bytes"
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"os"
	"strings"
)

func main() {
	body, _ := json.Marshal(map[string]any{
		"model":    "openai/gpt-5.2",
		"stream":   true,
		"messages": []map[string]string{{"role": "user", "content": "Hello!"}},
	})
	req, _ := http.NewRequest("POST", "https://openrouter.ai/api/v1/chat/completions", bytes.NewReader(body))
	req.Header.Set("Authorization", "Bearer "+os.Getenv("OPENROUTER_API_KEY"))
	req.Header.Set("Content-Type", "application/json")

	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		panic(err)
	}
	defer resp.Body.Close()
	if resp.StatusCode != http.StatusOK {
		b, _ := io.ReadAll(resp.Body)
		fmt.Fprintln(os.Stderr, resp.Status, string(b))
		os.Exit(1)
	}

	scanner := bufio.NewScanner(resp.Body)
	scanner.Buffer(make([]byte, 1024*1024), 1024*1024)
	for scanner.Scan() {
		line := scanner.Text()
		if !strings.HasPrefix(line, "data: ") {
			continue
		}
		data := strings.TrimPrefix(line, "data: ")
		if data == "[DONE]" {
			break
		}
		var chunk struct {
			Choices []struct {
				Delta struct {
					Content string `json:"content"`
				} `json:"delta"`
			} `json:"choices"`
		}
		if json.Unmarshal([]byte(data), &chunk) == nil && len(chunk.Choices) > 0 {
			fmt.Print(chunk.Choices[0].Delta.Content)
		}
	}
	fmt.Println()
}
```

### Server-Side Example

<CodeGroup>

```typescript title="TypeScript" lines
import { chat, toServerSentEventsResponse } from "@tanstack/ai";
import { openRouterText } from "@tanstack/ai-openrouter";

export async function POST(request: Request) {
  const { messages } = await request.json();

  const stream = chat({
    adapter: openRouterText("openai/gpt-5.2"),
    messages,
  });

  return toServerSentEventsResponse(stream);
}
```

```python title="Python" lines
import json
import os
from openrouter import OpenRouter

client = OpenRouter(api_key=os.environ["OPENROUTER_API_KEY"])


def post(request_body: str):
    messages = json.loads(request_body)["messages"]

    stream = client.chat.send(
        model="openai/gpt-5.2",
        messages=messages,
        stream=True,
    )

    # Server-Sent Events body, one event per chunk
    for chunk in stream:
        yield f"data: {chunk.model_dump_json(exclude_none=True)}\n\n"
    yield "data: [DONE]\n\n"


if __name__ == "__main__":
    body = json.dumps({"messages": [{"role": "user", "content": "Hello!"}]})
    for event in post(body):
        print(event, end="", flush=True)
```

```go title="Go" lines
package main

import (
	"bytes"
	"encoding/json"
	"io"
	"net/http"
	"net/http/httptest"
	"os"
	"strings"
)

// chatHandler streams an OpenRouter chat completion back to the client as SSE.
func chatHandler(w http.ResponseWriter, r *http.Request) {
	var in struct {
		Messages []map[string]any `json:"messages"`
	}
	if err := json.NewDecoder(r.Body).Decode(&in); err != nil {
		http.Error(w, err.Error(), http.StatusBadRequest)
		return
	}

	body, _ := json.Marshal(map[string]any{
		"model":    "openai/gpt-5.2",
		"stream":   true,
		"messages": in.Messages,
	})
	req, _ := http.NewRequestWithContext(r.Context(), "POST", "https://openrouter.ai/api/v1/chat/completions", bytes.NewReader(body))
	req.Header.Set("Authorization", "Bearer "+os.Getenv("OPENROUTER_API_KEY"))
	req.Header.Set("Content-Type", "application/json")

	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		http.Error(w, err.Error(), http.StatusBadGateway)
		return
	}
	defer resp.Body.Close()

	w.Header().Set("Content-Type", "text/event-stream")
	w.Header().Set("Cache-Control", "no-cache")
	w.WriteHeader(resp.StatusCode)
	flusher, _ := w.(http.Flusher)
	buf := make([]byte, 4096)
	for {
		n, err := resp.Body.Read(buf)
		if n > 0 {
			w.Write(buf[:n])
			if flusher != nil {
				flusher.Flush()
			}
		}
		if err != nil {
			return
		}
	}
}

func main() {
	// In production: http.HandleFunc("/api/chat", chatHandler); http.ListenAndServe(":8080", nil)
	srv := httptest.NewServer(http.HandlerFunc(chatHandler))
	defer srv.Close()

	resp, err := http.Post(srv.URL, "application/json",
		strings.NewReader(`{"messages":[{"role":"user","content":"Hello!"}]}`))
	if err != nil {
		panic(err)
	}
	defer resp.Body.Close()
	io.Copy(os.Stdout, resp.Body)
}
```

### Using Tools

<CodeGroup>

```typescript title="TypeScript" expandable lines
import { chat, toolDefinition } from "@tanstack/ai";
import { openRouterText } from "@tanstack/ai-openrouter";
import { z } from "zod";

const getWeatherDef = toolDefinition({
  name: "get_weather",
  description: "Get the current weather",
  inputSchema: z.object({
    location: z.string(),
  }),
});

const getWeather = getWeatherDef.server(async ({ location }) => {
  return { temperature: 72, conditions: "sunny" };
});

const messages = [{ role: "user" as const, content: "What's the weather in NYC?" }];

const stream = chat({
  adapter: openRouterText("openai/gpt-5.2"),
  threadId: "weather-demo",
  messages,
  tools: [getWeather],
});

for await (const chunk of stream as AsyncIterable<any>) {
  if ((chunk.type === "content" || chunk.type === "TEXT_MESSAGE_CONTENT") && chunk.delta) {
    process.stdout.write(chunk.delta);
  }
}
console.log();

```

```python title="Python" lines
import asyncio
import os
from pydantic import BaseModel
from openrouter_agent import OpenRouter, call_model, tool


class WeatherInput(BaseModel):
    location: str


class WeatherOutput(BaseModel):
    temperature: int
    conditions: str


get_weather = tool(
    name="get_weather",
    description="Get the current weather",
    input_schema=WeatherInput,
    output_schema=WeatherOutput,
    execute=lambda params, ctx: WeatherOutput(temperature=72, conditions="sunny"),
)


async def main() -> None:
    client = OpenRouter(api_key=os.environ["OPENROUTER_API_KEY"])

    result = call_model(
        client,
        {
            "model": "openai/gpt-5.2",
            "input": "What's the weather in NYC?",
            "tools": [get_weather],
        },
    )

    async for delta in result.get_text_stream():
        print(delta, end="", flush=True)
    print()


asyncio.run(main())

```

```go title="Go" lines
package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	"net/http"
	"os"
)

func getWeather(location string) map[string]any {
	return map[string]any{"temperature": 72, "conditions": "sunny"}
}

func send(payload map[string]any) map[string]any {
	body, _ := json.Marshal(payload)
	req, _ := http.NewRequest("POST", "https://openrouter.ai/api/v1/chat/completions", bytes.NewReader(body))
	req.Header.Set("Authorization", "Bearer "+os.Getenv("OPENROUTER_API_KEY"))
	req.Header.Set("Content-Type", "application/json")
	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		panic(err)
	}
	defer resp.Body.Close()
	var out map[string]any
	json.NewDecoder(resp.Body).Decode(&out)
	choices, ok := out["choices"].([]any)
	if !ok || len(choices) == 0 {
		panic(fmt.Sprintf("unexpected response: %v", out))
	}
	return choices[0].(map[string]any)["message"].(map[string]any)
}

func main() {
	tools := []any{map[string]any{
		"type": "function",
		"function": map[string]any{
			"name":        "get_weather",
			"description": "Get the current weather",
			"parameters": map[string]any{
				"type":       "object",
				"properties": map[string]any{"location": map[string]any{"type": "string"}},
				"required":   []string{"location"},
			},
		},
	}}
	messages := []any{map[string]any{"role": "user", "content": "What's the weather in NYC?"}}

	for {
		msg := send(map[string]any{"model": "openai/gpt-5.2", "messages": messages, "tools": tools})
		calls, _ := msg["tool_calls"].([]any)
		if len(calls) == 0 {
			fmt.Println(msg["content"])
			return
		}
		messages = append(messages, map[string]any{"role": "assistant", "content": msg["content"], "tool_calls": calls})
		for _, c := range calls {
			call := c.(map[string]any)
			fn := call["function"].(map[string]any)
			var args struct {
				Location string `json:"location"`
			}
			json.Unmarshal([]byte(fn["arguments"].(string)), &args)
			result, _ := json.Marshal(getWeather(args.Location))
			messages = append(messages, map[string]any{"role": "tool", "tool_call_id": call["id"], "content": string(result)})
		}
	}
}
```

#### Provider Ordering

Specify an explicit order of providers to try:

<CodeGroup>

```typescript title="TypeScript" lines
import { chat } from "@tanstack/ai";
import { openRouterText } from "@tanstack/ai-openrouter";

const stream = chat({
  adapter: openRouterText("anthropic/claude-sonnet-4.5"),
  threadId: "provider-order-demo",
  messages: [{ role: "user", content: "Hello!" }],
  modelOptions: {
```
