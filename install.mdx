## API key setup

You must set your OpenRouter API key as an environment variable. The API key is required for all requests.

**On macOS/Linux:**

```bash lines
export OPENROUTER_API_KEY="sk-or-v1-YOUR-API-KEY"
```

**On Windows (PowerShell):**

```powershell lines
$env:OPENROUTER_API_KEY="sk-or-v1-YOUR-API-KEY"
```

**On Windows (Command Prompt):**

```cmd lines
set OPENROUTER_API_KEY=sk-or-v1-YOUR-API-KEY
```

## Create a test script

Your task is to create a test script that makes a Responses API request and outputs the result.

**TypeScript - create `test_openrouter.ts`:**

<CodeGroup>

```typescript expandable lines
import { OpenRouter } from '@openrouter/sdk';

const openRouter = new OpenRouter({
  apiKey: process.env.OPENROUTER_API_KEY,
});

async function main() {
  const response = await openRouter.responses.send({
    responsesRequest: {
      model: 'openai/gpt-4o-mini',
      input: 'Say "Hello from OpenRouter!" and nothing else.',
      stream: false,
    },
  });

  if (!('output' in response)) {
    throw new Error('Expected a non-streaming response');
  }

  const message = response.output.find((item) => item.type === 'message');
  console.log(message?.content.find((part) => part.type === 'output_text')?.text);
}

main();
```

```python title="Python" lines
import os
from openrouter import OpenRouter

with OpenRouter(api_key=os.environ["OPENROUTER_API_KEY"]) as open_router:
    response = open_router.responses.send(
        model="openai/gpt-4o-mini",
        input='Say "Hello from OpenRouter!" and nothing else.',
        stream=False,
    )

    message = next(item for item in response.output if getattr(item, "type", None) == "message")
    print(next(part.text for part in message.content if getattr(part, "type", None) == "output_text"))

```

```go title="Go" lines
package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	"io"
	"log"
	"net/http"
	"os"
)

func main() {
	body, _ := json.Marshal(map[string]any{
		"model":  "openai/gpt-4o-mini",
		"input":  `Say "Hello from OpenRouter!" and nothing else.`,
		"stream": false,
	})

	req, _ := http.NewRequest("POST", "https://openrouter.ai/api/v1/responses", bytes.NewReader(body))
	req.Header.Set("Authorization", "Bearer "+os.Getenv("OPENROUTER_API_KEY"))
	req.Header.Set("Content-Type", "application/json")

	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		log.Fatal(err)
	}
	defer resp.Body.Close()

	if resp.StatusCode < 200 || resp.StatusCode > 299 {
		errBody, _ := io.ReadAll(resp.Body)
		log.Fatalf("request failed: %s: %s", resp.Status, errBody)
	}

	var result struct {
		Output []struct {
			Type    string `json:"type"`
			Content []struct {
				Type string `json:"type"`
				Text string `json:"text"`
			} `json:"content"`
		} `json:"output"`
	}
	if err := json.NewDecoder(resp.Body).Decode(&result); err != nil {
		log.Fatal(err)
	}

	for _, item := range result.Output {
		if item.Type != "message" {
			continue
		}
		for _, part := range item.Content {
			if part.Type == "output_text" {
				fmt.Println(part.Text)
				return
			}
		}
	}
}

```
