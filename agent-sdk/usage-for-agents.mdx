Give your AI coding assistant the knowledge to work with the OpenRouter Agent SDK by installing our official `openrouter-typescript-sdk` skill from the [OpenRouterTeam/skills](https://github.com/OpenRouterTeam/skills) repository.

The skill covers both the Agent SDK (`@openrouter/agent`) and the Client SDKs (`@openrouter/sdk`). When working with the Agent SDK, your AI assistant will focus on the **agent features**: `callModel`, `tool()` definitions, stop conditions, streaming, and multi-turn conversations.

## Quick Start

### Claude Code

```bash lines
/plugin marketplace add OpenRouterTeam/skills
/plugin install openrouter@openrouter
```

### Cursor

Add via **Settings > Rules > Add Rule > Remote Rule (Github)** with `OpenRouterTeam/skills`.

### GitHub CLI

Requires [GitHub CLI](https://cli.github.com/) v2.90.0+. Works with Claude Code, Cursor, OpenCode, Codex, Gemini CLI, Windsurf, and [many more agents](https://cli.github.com/manual/gh_skill_install):

```bash lines
gh skill install OpenRouterTeam/skills openrouter-typescript-sdk
```

## What the Skill Provides

Once installed, your AI coding assistant will have knowledge of:

- **callModel API** - The recommended approach for making AI model calls with full type safety and streaming support
- **Tool Definitions** - Creating tools with the `tool()` helper and Zod schemas
- **Stop Conditions** - Controlling agent loop termination with `stepCountIs`, `maxCost`, and more
- **Streaming** - Real-time token output within agent steps
- **Multi-turn Conversations** - Managing conversation state across turns
- **Dynamic Parameters** - Changing model, temperature, or tools between turns

## Example Usage

After installing the skill, your AI assistant can help you with tasks like the ones below. The skill is `openrouter-typescript-sdk`, so the TypeScript tabs show what the assistant writes. The Python and Go tabs are independent equivalents for readers working in those languages; the skill does not teach them.

**"Help me set up an OpenRouter agent"**

The assistant will know to use:

<CodeGroup>

```typescript lines
import { OpenRouter, callModel } from '@openrouter/agent';

const openrouter = new OpenRouter({
  apiKey: process.env.OPENROUTER_API_KEY,
});

const result = callModel(openrouter, {
  model: '~anthropic/claude-sonnet-latest',
  input: 'Hello!'
});

const text = await result.getText();
console.log(text);

```

```python title="Python" lines
import asyncio
import os
from openrouter_agent import OpenRouter, call_model


async def main() -> None:
    openrouter = OpenRouter(api_key=os.environ["OPENROUTER_API_KEY"])

    result = call_model(
        openrouter,
        {
            "model": "~anthropic/claude-sonnet-latest",
            "input": "Hello!",
        },
    )

    text = await result.get_text()
    print(text)


asyncio.run(main())

```

```go title="Go" lines
package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	"log"
	"net/http"
	"os"
)

func main() {
	body, _ := json.Marshal(map[string]any{
		"model":    "~anthropic/claude-sonnet-latest",
		"messages": []map[string]string{{"role": "user", "content": "Hello!"}},
	})

	req, _ := http.NewRequest("POST", "https://openrouter.ai/api/v1/chat/completions", bytes.NewReader(body))
	req.Header.Set("Authorization", "Bearer "+os.Getenv("OPENROUTER_API_KEY"))
	req.Header.Set("Content-Type", "application/json")

	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		log.Fatal(err)
	}
	defer resp.Body.Close()

	var result struct {
		Choices []struct {
			Message struct {
				Content string `json:"content"`
			} `json:"message"`
		} `json:"choices"`
	}
	if err := json.NewDecoder(resp.Body).Decode(&result); err != nil {
		log.Fatal(err)
	}
	if len(result.Choices) == 0 {
		log.Fatalf("no choices returned (status %d)", resp.StatusCode)
	}

	fmt.Println(result.Choices[0].Message.Content)
}

```

**"Add a tool to my agent"**

The assistant understands the tool pattern:

<CodeGroup>

```typescript lines
import { OpenRouter, callModel, tool } from '@openrouter/agent';
import { z } from 'zod';

const openrouter = new OpenRouter({
  apiKey: process.env.OPENROUTER_API_KEY,
});

const searchTool = tool({
  name: 'search',
  description: 'Search the web',
  inputSchema: z.object({ query: z.string() }),
  execute: async ({ query }) => {
    return { results: ['...'] };
  },
});

const result = callModel(openrouter, {
  model: '~anthropic/claude-sonnet-latest',
  input: 'Search for TypeScript best practices',
  tools: [searchTool],
});

const text = await result.getText();
console.log(text);

```

```python title="Python" lines
import asyncio
import os
from pydantic import BaseModel
from openrouter_agent import OpenRouter, call_model, tool


class SearchInput(BaseModel):
    query: str


def search(params: SearchInput, ctx) -> dict:
    return {"results": ["..."]}


search_tool = tool(
    name="search",
    description="Search the web",
    input_schema=SearchInput,
    execute=search,
)


async def main() -> None:
    openrouter = OpenRouter(api_key=os.environ["OPENROUTER_API_KEY"])

    result = call_model(
        openrouter,
        {
            "model": "~anthropic/claude-sonnet-latest",
            "input": "Search for TypeScript best practices",
            "tools": [search_tool],
        },
    )

    text = await result.get_text()
    print(text)


asyncio.run(main())

```

```go title="Go" lines
package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	"log"
	"net/http"
	"os"
)

type toolCall struct {
	ID       string `json:"id"`
	Type     string `json:"type"`
	Function struct {
		Name      string `json:"name"`
		Arguments string `json:"arguments"`
	} `json:"function"`
}

type message struct {
	Role       string     `json:"role"`
	Content    string     `json:"content,omitempty"`
	ToolCalls  []toolCall `json:"tool_calls,omitempty"`
	ToolCallID string     `json:"tool_call_id,omitempty"`
}

func search(query string) map[string]any {
	return map[string]any{"results": []string{"..."}}
}

func main() {
	tools := []map[string]any{{
		"type": "function",
		"function": map[string]any{
			"name":        "search",
			"description": "Search the web",
			"parameters": map[string]any{
				"type":       "object",
				"properties": map[string]any{"query": map[string]string{"type": "string"}},
				"required":   []string{"query"},
			},
		},
	}}
	messages := []message{{Role: "user", Content: "Search for TypeScript best practices"}}

	for i := 0; i < 5; i++ {
		body, _ := json.Marshal(map[string]any{
			"model":    "~anthropic/claude-sonnet-latest",
			"messages": messages,
			"tools":    tools,
		})
		req, _ := http.NewRequest("POST", "https://openrouter.ai/api/v1/chat/completions", bytes.NewReader(body))
		req.Header.Set("Authorization", "Bearer "+os.Getenv("OPENROUTER_API_KEY"))
		req.Header.Set("Content-Type", "application/json")

		resp, err := http.DefaultClient.Do(req)
		if err != nil {
			log.Fatal(err)
		}
		var result struct {
			Choices []struct {
				Message message `json:"message"`
			} `json:"choices"`
		}
		err = json.NewDecoder(resp.Body).Decode(&result)
		resp.Body.Close()
		if err != nil {
			log.Fatal(err)
		}
		if len(result.Choices) == 0 {
			log.Fatalf("no choices returned (status %d)", resp.StatusCode)
		}

		msg := result.Choices[0].Message
		if len(msg.ToolCalls) == 0 {
			fmt.Println(msg.Content)
			return
		}
		messages = append(messages, msg)
		for _, tc := range msg.ToolCalls {
			var args struct {
				Query string `json:"query"`
			}
			if err := json.Unmarshal([]byte(tc.Function.Arguments), &args); err != nil || args.Query == "" {
				log.Fatalf("bad arguments for %s: %s", tc.Function.Name, tc.Function.Arguments)
			}
			out, _ := json.Marshal(search(args.Query))
			messages = append(messages, message{Role: "tool", ToolCallID: tc.ID, Content: string(out)})
		}
	}
	log.Fatal("too many tool-call rounds")
}

```
