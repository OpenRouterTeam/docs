## Overview

This example shows how to build encapsulated, self-managing tools that inject domain-specific context into conversations. When a skill is loaded, it automatically enriches subsequent turns with specialized instructions.

```typescript expandable lines
import { OpenRouter, tool } from '@openrouter/agent';
import { readFileSync, existsSync, readdirSync } from 'fs';
import path from 'path';
import { z } from 'zod';

const openrouter = new OpenRouter({
  apiKey: process.env.OPENROUTER_API_KEY,
});

const SKILLS_DIR = path.join(process.env.HOME || '~', '.claude', 'skills');

// List available skills
const listAvailableSkills = (): string[] => {
  if (!existsSync(SKILLS_DIR)) return [];
  return readdirSync(SKILLS_DIR, { withFileTypes: true })
    .filter((dirent) => dirent.isDirectory())
    .filter((dirent) => existsSync(path.join(SKILLS_DIR, dirent.name, 'SKILL.md')))
    .map((dirent) => dirent.name);
};

const skillsTool = tool({
  name: 'Skill',
  description: `Load a specialized skill to enhance the assistant's capabilities.
Available skills: ${listAvailableSkills().join(', ') || 'none configured'}
Each skill provides domain-specific instructions and capabilities.`,

  inputSchema: z.object({
    type: z.string().describe("The skill type to load (e.g., 'pdf-processing')"),
  }),

  outputSchema: z.string(),

  // This is where the magic happens - modify context for next turn
  nextTurnParams: {
    input: (params, context) => {
      // Prevent duplicate skill loading
      const skillMarker = `[Skill: ${params.type}]`;
      if (JSON.stringify(context.input).includes(skillMarker)) {
        return context.input;
      }

      // The skill name comes from the model, so only load skills that exist under SKILLS_DIR
      if (!listAvailableSkills().includes(params.type)) {
        return context.input;
      }

      // Load the skill's instructions
      const skillPath = path.join(SKILLS_DIR, params.type, 'SKILL.md');

      const skill = readFileSync(skillPath, 'utf-8');
      const skillDir = path.join(SKILLS_DIR, params.type);

      // Inject skill context into the conversation
      const currentInput = Array.isArray(context.input) ? context.input : [context.input];

      return [
        ...currentInput,
        {
          role: 'user',
          content: `${skillMarker}
Base directory for this skill: ${skillDir}

${skill}`,
        },
      ];
    },
  },

  execute: async (params, context) => {
    const skillMarker = `[Skill: ${params.type}]`;

    // Check if already loaded
    if (JSON.stringify(context?.turnRequest?.input || []).includes(skillMarker)) {
      return `Skill ${params.type} is already loaded`;
    }

    const available = listAvailableSkills();
    if (!available.includes(params.type)) {
      return `Skill "${params.type}" not found. Available skills: ${available.join(', ') || 'none'}`;
    }

    return `Launching skill ${params.type}`;
  },
});
```

```python title="Python" lines
import asyncio
import json
import os
from pathlib import Path

from pydantic import BaseModel, Field
from openrouter_agent import OpenRouter, call_model, tool

client = OpenRouter(api_key=os.environ["OPENROUTER_API_KEY"])

SKILLS_DIR = Path.home() / ".claude" / "skills"


# List available skills
def list_available_skills() -> list[str]:
    if not SKILLS_DIR.exists():
        return []
    return [d.name for d in SKILLS_DIR.iterdir() if d.is_dir() and (d / "SKILL.md").exists()]


class SkillInput(BaseModel):
    type: str = Field(description="The skill type to load (e.g., 'pdf-processing')")


# This is where the magic happens - modify context for next turn
# next_turn_params functions receive the raw tool-call arguments as a dict
def next_input(params, context):
    skill_type = params["type"]
    current = context["input"]
    # Prevent duplicate skill loading
    marker = f"[Skill: {skill_type}]"
    if marker in json.dumps(current, default=str):
        return current

    # The skill name comes from the model, so only load skills that exist under SKILLS_DIR
    if skill_type not in list_available_skills():
        return current

    # Load the skill's instructions
    skill = (SKILLS_DIR / skill_type / "SKILL.md").read_text(encoding="utf-8")
    skill_dir = SKILLS_DIR / skill_type

    # Inject skill context into the conversation
    items = current if isinstance(current, list) else [current]
    return [
        *items,
        {
            "role": "user",
            "content": f"{marker}\nBase directory for this skill: {skill_dir}\n\n{skill}",
        },
    ]


def execute(params, ctx):
    marker = f"[Skill: {params.type}]"

    # Check if already loaded
    turn_request = (ctx or {}).get("turn_request") or {}
    if marker in json.dumps(turn_request.get("input", []), default=str):
        return f"Skill {params.type} is already loaded"

    available = list_available_skills()
    if params.type not in available:
        return f'Skill "{params.type}" not found. Available skills: {", ".join(available) or "none"}'

    return f"Launching skill {params.type}"


skills_tool = tool(
    name="Skill",
    description=f"""Load a specialized skill to enhance the assistant's capabilities.
Available skills: {', '.join(list_available_skills()) or 'none configured'}
Each skill provides domain-specific instructions and capabilities.""",
    input_schema=SkillInput,
    next_turn_params={"input": next_input},
    execute=execute,
)


async def main() -> None:
    skills = list_available_skills()
    print("Available skills:", ", ".join(skills) or "none configured")
    skill_name = skills[0] if skills else "pdf-processing"

    result = call_model(
        client,
        {
            "model": "openai/gpt-4o",
            "input": f"Use the Skill tool to load the '{skill_name}' skill, then briefly summarize what it lets you do.",
            "tools": [skills_tool],
        },
    )

    print(await result.get_text())


asyncio.run(main())
```

```go title="Go" lines
package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"os"
	"path/filepath"
	"slices"
	"strings"
)

var skillsDir = filepath.Join(os.Getenv("HOME"), ".claude", "skills")

type toolCall struct {
	ID       string `json:"id"`
	Type     string `json:"type"`
	Function struct {
		Name      string `json:"name"`
		Arguments string `json:"arguments"`
	} `json:"function"`
}

type message struct {
	Role       string     `json:"role"`
	Content    any        `json:"content"`
	ToolCalls  []toolCall `json:"tool_calls,omitempty"`
	ToolCallID string     `json:"tool_call_id,omitempty"`
}

// List available skills
func listAvailableSkills() []string {
	entries, err := os.ReadDir(skillsDir)
	if err != nil {
		return nil
	}
	var names []string
	for _, e := range entries {
		if _, err := os.Stat(filepath.Join(skillsDir, e.Name(), "SKILL.md")); e.IsDir() && err == nil {
			names = append(names, e.Name())
		}
	}
	return names
}

func orNone(names []string, fallback string) string {
	if len(names) == 0 {
		return fallback
	}
	return strings.Join(names, ", ")
}

// executeSkill returns the tool output plus a message to inject into the next turn.
func executeSkill(skillType string, messages []message) (string, *message) {
	marker := fmt.Sprintf("[Skill: %s]", skillType)
	history, _ := json.Marshal(messages)
	if strings.Contains(string(history), marker) {
		return "Skill " + skillType + " is already loaded", nil
	}
	// The skill name comes from the model, so only load skills that exist under skillsDir
	available := listAvailableSkills()
	if !slices.Contains(available, skillType) {
		return fmt.Sprintf("Skill %q not found. Available skills: %s", skillType, orNone(available, "none")), nil
	}
	skillDir := filepath.Join(skillsDir, skillType)
	skill, err := os.ReadFile(filepath.Join(skillDir, "SKILL.md"))
	if err != nil {
		return fmt.Sprintf("Skill %q could not be read: %v", skillType, err), nil
	}
	return "Launching skill " + skillType, &message{
		Role:    "user",
		Content: fmt.Sprintf("%s\nBase directory for this skill: %s\n\n%s", marker, skillDir, skill),
	}
}

func main() {
	skillTool := map[string]any{
		"type": "function",
		"function": map[string]any{
			"name": "Skill",
			"description": fmt.Sprintf("Load a specialized skill to enhance the assistant's capabilities.\nAvailable skills: %s\nEach skill provides domain-specific instructions and capabilities.", orNone(listAvailableSkills(), "none configured")),
			"parameters": map[string]any{
				"type": "object",
				"properties": map[string]any{
					"type": map[string]any{"type": "string", "description": "The skill type to load (e.g., 'pdf-processing')"},
				},
				"required": []string{"type"},
			},
		},
	}

	messages := []message{{Role: "user", Content: "Load the pdf-processing skill, then tell me what it lets you do."}}

	for i := 0; i < 5; i++ {
		body, _ := json.Marshal(map[string]any{"model": "openai/gpt-4o-mini", "messages": messages, "tools": []any{skillTool}})
		req, _ := http.NewRequest("POST", "https://openrouter.ai/api/v1/chat/completions", bytes.NewReader(body))
		req.Header.Set("Authorization", "Bearer "+os.Getenv("OPENROUTER_API_KEY"))
		req.Header.Set("Content-Type", "application/json")
		resp, err := http.DefaultClient.Do(req)
		if err != nil {
			panic(err)
		}
		if resp.StatusCode != http.StatusOK {
			b, _ := io.ReadAll(resp.Body)
			panic(fmt.Sprintf("HTTP %d: %s", resp.StatusCode, b))
		}
		var out struct {
			Choices []struct {
				Message message `json:"message"`
			} `json:"choices"`
		}
		err = json.NewDecoder(resp.Body).Decode(&out)
		resp.Body.Close()
		if err != nil || len(out.Choices) == 0 {
			panic(fmt.Sprintf("bad response: %v", err))
		}

		msg := out.Choices[0].Message
		messages = append(messages, msg)
		if len(msg.ToolCalls) == 0 {
			fmt.Println(msg.Content)
			return
		}

		// Run each tool call, then inject loaded skill instructions for the next turn
		var injected []message
		for _, tc := range msg.ToolCalls {
			var args struct {
				Type string `json:"type"`
			}
			json.Unmarshal([]byte(tc.Function.Arguments), &args)
			result, inject := executeSkill(args.Type, messages)
			messages = append(messages, message{Role: "tool", ToolCallID: tc.ID, Content: result})
			if inject != nil {
				injected = append(injected, *inject)
			}
		}
		messages = append(messages, injected...)
	}
}
```

```typescript lines
import { join } from 'node:path';

    .map((e) => e.name);
}

const available = listAvailableSkills();

const skillsTool = tool({
  name: 'Skill',
  description: `Load a specialized skill to enhance the assistant's capabilities.\nAvailable skills: ${available.join(', ') || 'none configured'}\nEach skill provides domain-specific instructions and capabilities.`,
  inputSchema: z.object({
    type: z.string().describe("The skill type to load (e.g., 'pdf-processing')"),
  }),
```

## Example Skill File

Create `~/.claude/skills/pdf-processing/SKILL.md`:

```markdown expandable lines
# PDF Processing Skill

You are now equipped with PDF processing capabilities.

## Available Tools
When processing PDFs, you have access to:
- `extract_text`: Extract all text from a PDF
- `extract_tables`: Extract tables as structured data
- `extract_images`: Extract embedded images
- `split_pdf`: Split PDF into individual pages

## Best Practices
1. Always check PDF file size before processing
2. For large PDFs (>50 pages), process in chunks
3. OCR may be needed for scanned documents
4. Tables may span multiple pages - handle accordingly

## Output Formats
- Text: Plain text or markdown
- Tables: JSON, CSV, or markdown tables
- Images: PNG with sequential naming

## Error Handling
- If a PDF is encrypted, request the password
- If OCR fails, suggest alternative approaches
- Report page numbers for any extraction errors
```

## Extended: Multi-Skill Loader

Load multiple skills in a single call:

```typescript expandable lines
const multiSkillLoader = tool({
  name: 'load_skills',
  description: 'Load multiple skills at once for complex tasks',

  inputSchema: z.object({
    skills: z.array(z.string()).describe('Array of skill names to load'),
  }),

  outputSchema: z.object({
    loaded: z.array(z.string()),
    failed: z.array(
      z.object({
        name: z.string(),
        reason: z.string(),
      })
    ),
  }),

  nextTurnParams: {
    input: (params, context) => {
      let newInput = Array.isArray(context.input) ? context.input : [context.input];

      for (const skillName of params.skills) {
        const skillMarker = `[Skill: ${skillName}]`;

        // Skip if already loaded
        if (JSON.stringify(newInput).includes(skillMarker)) {
          continue;
        }

        const skillPath = path.join(SKILLS_DIR, skillName, 'SKILL.md');
        if (!existsSync(skillPath)) {
          continue;
        }

        const skillContent = readFileSync(skillPath, 'utf-8');
        const skillDir = path.join(SKILLS_DIR, skillName);

        newInput = [
          ...newInput,
          {
            role: 'user',
            content: `${skillMarker}
Base directory: ${skillDir}

${skillContent}`,
          },
        ];
      }

      return newInput;
    },
  },

  execute: async ({ skills }) => {
    const loaded: string[] = [];
    const failed: Array<{ name: string; reason: string }> = [];

    for (const skill of skills) {
      const skillPath = path.join(SKILLS_DIR, skill, 'SKILL.md');
      if (existsSync(skillPath)) {
        loaded.push(skill);
      } else {
        failed.push({ name: skill, reason: 'Skill not found' });
      }
    }

    return { loaded, failed };
  },
});

// Usage
const result = openrouter.callModel({
  model: 'anthropic/claude-sonnet-4.5',
  input: 'I need to analyze a PDF report and create visualizations',
  tools: [multiSkillLoader],
});
// Model might call: load_skills({ skills: ['pdf-processing', 'data-analysis'] })
```

```typescript expandable lines
const configurableSkillLoader = tool({
  name: 'configure_skill',
  description: 'Load a skill with custom configuration options',

  inputSchema: z.object({
    skillName: z.string(),
    options: z
      .object({
        verbosity: z.enum(['minimal', 'normal', 'detailed']).default('normal'),
        strictMode: z.boolean().default(false),
        outputFormat: z.enum(['json', 'markdown', 'plain']).default('markdown'),
      })
      .optional(),
  }),

  outputSchema: z.object({
    status: z.enum(['loaded', 'already_loaded', 'not_found']),
    message: z.string(),
    configuration: z.record(z.unknown()).optional(),
  }),

  nextTurnParams: {
    input: (params, context) => {
      const skillMarker = `[Skill: ${params.skillName}]`;
      if (JSON.stringify(context.input).includes(skillMarker)) {
        return context.input;
      }

      const skillPath = path.join(SKILLS_DIR, params.skillName, 'SKILL.md');
      if (!existsSync(skillPath)) {
        return context.input;
      }

      const skillContent = readFileSync(skillPath, 'utf-8');
      const options = params.options || {};

      // Build configuration header
      const configHeader = `
## Skill Configuration
- Verbosity: ${options.verbosity || 'normal'}
- Strict Mode: ${options.strictMode || false}
- Output Format: ${options.outputFormat || 'markdown'}
`;

      // A plain string input must become a message item before appending
      const currentInput =
        typeof context.input === 'string'
          ? [{ role: 'user' as const, content: context.input }]
          : (context.input ?? []);

      return [
        ...currentInput,
        {
          role: 'user' as const,
          content: `${skillMarker}
${configHeader}

${skillContent}`,
        },
      ];
    },

    // Adjust model behavior based on skill
    temperature: (params, context) => {
      // Lower temperature for strict mode
      if (params.options?.strictMode) {
        return 0.3;
      }
      return context.temperature;
    },
  },

  execute: async ({ skillName, options }) => {
    const skillPath = path.join(SKILLS_DIR, skillName, 'SKILL.md');

    if (!existsSync(skillPath)) {
      return {
        status: 'not_found' as const,
        message: `Skill "${skillName}" not found`,
      };
    }

    return {
      status: 'loaded' as const,
      message: `Skill "${skillName}" loaded with configuration`,
      configuration: options || {},
    };
  },
});
```

```python title="Python" lines
class SkillOptions(BaseModel):
    verbosity: Literal["minimal", "normal", "detailed"] = "normal"
    strict_mode: bool = False
    output_format: Literal["json", "markdown", "plain"] = "markdown"


class ConfigureSkillInput(BaseModel):
    skill_name: str
    options: SkillOptions | None = None


class ConfigureSkillOutput(BaseModel):
    status: Literal["loaded", "already_loaded", "not_found"]
    message: str
    configuration: dict[str, Any] | None = None


# The skill name comes from the model, so accept a plain directory name only
def is_skill_name(name: str) -> bool:
    return name not in ("", ".", "..") and os.path.basename(name) == name


def next_input(params, context):
    # next_turn_params functions receive the raw tool-call arguments (a dict)
    params = ConfigureSkillInput.model_validate(params)
    skill_marker = f"[Skill: {params.skill_name}]"
    current = context.get("input")
    if skill_marker in json.dumps(current, default=str):
        return current

    skill_path = os.path.join(SKILLS_DIR, params.skill_name, "SKILL.md")
    if not is_skill_name(params.skill_name) or not os.path.exists(skill_path):
        return current

    with open(skill_path, encoding="utf-8") as f:
        skill_content = f.read()
    options = params.options or SkillOptions()

    # Build configuration header
    config_header = f"""
## Skill Configuration
- Verbosity: {options.verbosity}
- Strict Mode: {options.strict_mode}
- Output Format: {options.output_format}
"""

    # A plain string input must become a message item before appending
    if isinstance(current, str):
        current = [{"role": "user", "content": current}]

    return [
        *(current or []),
        {"role": "user", "content": f"{skill_marker}\n{config_header}\n\n{skill_content}"},
    ]


# Adjust model behavior based on skill
def next_temperature(params, context):
    params = ConfigureSkillInput.model_validate(params)
    # Lower temperature for strict mode
    if params.options and params.options.strict_mode:
        return 0.3
    return context.get("temperature")

    return ConfigureSkillOutput(
        status="loaded",
        message=f'Skill "{params.skill_name}" loaded with configuration',
        configuration=params.options.model_dump() if params.options else {},
```

```go title="Go" lines
func nextTurnInput(args configureSkillArgs, messages []message) []message {
	skillMarker := fmt.Sprintf("[Skill: %s]", args.SkillName)
	if b, _ := json.Marshal(messages); strings.Contains(string(b), skillMarker) {

	}
	if !slices.Contains(listAvailableSkills(), args.SkillName) {
		return messages
	}
	skillContent, err := os.ReadFile(filepath.Join(skillsDir, args.SkillName, "SKILL.md"))
	if err != nil {
		return messages
	}
	opts := skillOptions{}
	if args.Options != nil {
		opts = *args.Options
	}
	if opts.Verbosity == "" {
		opts.Verbosity = "normal"
	}
	if opts.OutputFormat == "" {

	}

		opts.Verbosity, opts.StrictMode, opts.OutputFormat)
```

```typescript expandable lines
const skillDiscoveryTool = tool({

    for (const skillName of availableSkills) {
      const skillPath = path.join(SKILLS_DIR, skillName, 'SKILL.md');
      const content = readFileSync(skillPath, 'utf-8');

      // Extract first paragraph as description
      const lines = content.split('\n').filter((l) => l.trim());
      const description = lines.find((l) => !l.startsWith('#')) || 'No description';

      // Check for config file
      const configPath = path.join(SKILLS_DIR, skillName, 'config.json');
      const hasConfig = existsSync(configPath);

      skills.push({
        name: skillName,
        description: description.slice(0, 100),
        hasConfig,
      });
    }

    return {
      skills,
      totalCount: skills.length,
    };
  },
```
