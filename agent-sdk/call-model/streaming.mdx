## Text Streaming

### getTextStream()

Stream text content as it's generated:

<CodeGroup>

```typescript lines
import { OpenRouter } from '@openrouter/agent';

const openrouter = new OpenRouter({
  apiKey: process.env.OPENROUTER_API_KEY,
});

const result = openrouter.callModel({
  model: 'openai/gpt-5-nano',
  input: 'Write a short poem about the ocean.',
});

for await (const delta of result.getTextStream()) {
  process.stdout.write(delta);
}
```

```python title="Python" lines
import asyncio
import os

from openrouter_agent import OpenRouter, call_model


async def main() -> None:
    client = OpenRouter(api_key=os.environ["OPENROUTER_API_KEY"])

    result = call_model(
        client,
        {
            "model": "openai/gpt-5-nano",
            "input": "Write a short poem about the ocean.",
        },
    )

    async for delta in result.get_text_stream():
        print(delta, end="", flush=True)
    print()


asyncio.run(main())

```

```go title="Go" lines

	"bufio"
	"bytes"
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"os"
	"strings"
)

func main() {
	body, _ := json.Marshal(map[string]any{
		"model":  "openai/gpt-5-nano",
		"input":  "Write a short poem about the ocean.",
		"stream": true,
	})

	req, _ := http.NewRequest("POST", "https://openrouter.ai/api/v1/responses", bytes.NewReader(body))
	req.Header.Set("Authorization", "Bearer "+os.Getenv("OPENROUTER_API_KEY"))
	req.Header.Set("Content-Type", "application/json")

	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		panic(err)
	}
	defer resp.Body.Close()
	if resp.StatusCode != http.StatusOK {
		b, _ := io.ReadAll(resp.Body)
		panic(fmt.Sprintf("%s: %s", resp.Status, b))
	}

	scanner := bufio.NewScanner(resp.Body)
	scanner.Buffer(make([]byte, 1024*1024), 1024*1024)
	for scanner.Scan() {
		data, ok := strings.CutPrefix(scanner.Text(), "data: ")
		if !ok || data == "[DONE]" {
			continue
		}
		var event struct {
			Type  string `json:"type"`
			Delta string `json:"delta"`
		}
		if json.Unmarshal([]byte(data), &event) == nil && event.Type == "response.output_text.delta" {
			fmt.Print(event.Delta)
		}
	}
	fmt.Println()
}
```

## Reasoning Streaming

### getReasoningStream()

For models that support reasoning (like o3 or Claude with thinking), stream the reasoning process:

<CodeGroup>

```typescript lines
const result = openrouter.callModel({
  model: 'openai/o3',
  input: 'Solve this step by step: If x + 5 = 12, what is x?',
});

console.log('Reasoning:');
for await (const delta of result.getReasoningStream()) {
  process.stdout.write(delta);
}

console.log('\n\nFinal answer:');
const text = await result.getText();
console.log(text);
```

```python title="Python" lines
import asyncio
import os

from openrouter_agent import OpenRouter, call_model


async def main() -> None:
    client = OpenRouter(api_key=os.environ["OPENROUTER_API_KEY"])

    result = call_model(
        client,
        {
            "model": "openai/o3",
            "input": "Solve this step by step: If x + 5 = 12, what is x?",
            "reasoning": {"effort": "medium", "summary": "auto"},
        },
    )

    print("Reasoning:")
    async for delta in result.get_reasoning_stream():
        print(delta, end="", flush=True)

    print("\n\nFinal answer:")
    text = await result.get_text()
    print(text)


asyncio.run(main())

```

## Items Streaming

### getItemsStream()

Stream complete items as they update. This is the **recommended way** to handle streaming when you need structured access to all output types (messages, tool calls, reasoning, etc.). See [Working with Items](/agent-sdk/call-model/items) for the full paradigm explanation.

<CodeGroup>

```typescript expandable lines
import type { StreamableOutputItem } from '@openrouter/agent';

const result = openrouter.callModel({
  model: '~anthropic/claude-sonnet-latest',
  input: 'Hello!',
  tools: [myTool],
});

for await (const item of result.getItemsStream()) {
  switch (item.type) {
    case 'message':
      console.log('Message:', item.content);
      break;
    case 'function_call':
      console.log('Tool call:', item.name, item.arguments);
      break;
    case 'reasoning':
      console.log('Thinking:', item.summary);
      break;
    case 'function_call_output':
      console.log('Tool result:', item.output);
      break;
  }
}
```

```python title="Python" lines
import asyncio
import os

from pydantic import BaseModel
from openrouter_agent import OpenRouter, call_model, tool


class WeatherInput(BaseModel):
    location: str


class WeatherOutput(BaseModel):
    temperature: int
    condition: str
    location: str


my_tool = tool(
    name="get_weather",
    description="Get the current weather for a location",
    input_schema=WeatherInput,
    output_schema=WeatherOutput,
    execute=lambda params, ctx: WeatherOutput(
        temperature=72, condition="sunny", location=params.location
    ),
)


async def main() -> None:
    client = OpenRouter(api_key=os.environ["OPENROUTER_API_KEY"])

    result = call_model(
        client,
        {
            "model": "~anthropic/claude-sonnet-latest",
            "input": "Hello!",
            "tools": [my_tool],
        },
    )

    async for item in result.get_items_stream():
        data = item if isinstance(item, dict) else item.model_dump()
        match data["type"]:
            case "message":
                print("Message:", data.get("content"))
            case "function_call":
                print("Tool call:", data.get("name"), data.get("arguments"))
            case "reasoning":
                print("Thinking:", data.get("summary"))
            case "function_call_output":
                print("Tool result:", data.get("output"))


asyncio.run(main())

```

```go title="Go" lines

	panic(err)

	b, _ := io.ReadAll(resp.Body)
	panic(fmt.Sprintf("%s: %s", resp.Status, b))
}

scanner.Buffer(make([]byte, 1024*1024), 1024*1024)
for scanner.Scan() {
	data, ok := strings.CutPrefix(scanner.Text(), "data: ")
	if !ok || data == "[DONE]" {
		continue
	}
	var event struct {
		Type  string `json:"type"`
		Delta string `json:"delta"`
		Item  struct {
			Type      string          `json:"type"`
			Name      string          `json:"name"`
			Arguments string          `json:"arguments"`
			Content   json.RawMessage `json:"content"`
			Summary   json.RawMessage `json:"summary"`
		} `json:"item"`
	}
	if json.Unmarshal([]byte(data), &event) != nil {
		continue
	}
	switch event.Type {
	case "response.failed", "error":
		panic("stream failed: " + data)
	case "response.output_item.added":
		text.Reset()
	case "response.output_text.delta":
		text.WriteString(event.Delta)
		fmt.Println("Message:", text.String())
	case "response.output_item.done":
		switch event.Item.Type {
		case "message":
			fmt.Println("Message:", string(event.Item.Content))
		case "function_call":
			fmt.Println("Tool call:", event.Item.Name, event.Item.Arguments)
		case "reasoning":
			fmt.Println("Thinking:", string(event.Item.Summary))
		}
	}
}
if err := scanner.Err(); err != nil {
	panic(err) // the stream ended early: what was printed is incomplete
}
```

**Key insight**: Each iteration yields a **complete item** with the same ID but updated content. Replace items by ID rather than accumulating deltas.

This stream yields all item types:

| Type                    | Description                        |
| ----------------------- | ---------------------------------- |
| `message`               | Assistant text responses           |
| `function_call`         | Tool invocations with arguments    |
| `reasoning`             | Model thinking (extended thinking) |
| `web_search_call`       | Web search operations              |
| `file_search_call`      | File search operations             |
| `image_generation_call` | Image generation operations        |
| `function_call_output`  | Results from executed tools        |

## Full Event Streaming

### getFullResponsesStream()

Stream all response events including tool preliminary results:

<CodeGroup>

```typescript expandable lines
const result = callModel(openrouter, {
  model: 'openai/gpt-5-nano',
  input: 'Search for documents about TypeScript generics',
  tools: [searchTool] as const, // Generator tool with eventSchema
});

for await (const event of result.getFullResponsesStream()) {
  switch (event.type) {
    case 'response.output_text.delta':
      process.stdout.write(event.delta);
      break;
    case 'response.function_call_arguments.delta':
      console.log('Tool argument delta:', event.delta);
      break;
    case 'response.completed':
      console.log('Response complete');
      break;
    case 'tool.preliminary_result':
      // Intermediate progress from generator tools
      console.log('Progress:', event.result);
      break;
    case 'tool.result':
      // Final result when tool execution completes
      console.log('Tool completed:', event.toolCallId);
      console.log('Result:', event.result);
      // Access any preliminary results that were emitted
      if (event.preliminaryResults) {
        console.log('Preliminary results:', event.preliminaryResults);
      }
      break;
  }
}
```

```python title="Python" lines
async def main():
    result = call_model(openrouter, {
        "model": "openai/gpt-5-nano",
        "input": "Search for documents about TypeScript generics",
        "tools": [search_tool],  # Generator tool with event_schema
    })

    async for event in result.get_full_responses_stream():
        e = event if isinstance(event, dict) else event.model_dump()
        match e.get("type"):
            case "response.output_text.delta":
                print(e["delta"], end="", flush=True)
            case "response.function_call_arguments.delta":
                print("Tool argument delta:", e["delta"])
            case "response.completed":
                print("Response complete")
            case "tool.preliminary_result":
                # Intermediate progress from generator tools
                print("Progress:", e.get("result"))
            case "tool.result":
                # Final result when tool execution completes
                print("Tool completed:", e.get("tool_call_id"))
                print("Result:", e.get("result"))
                # Access any preliminary results that were emitted
                if e.get("preliminary_results"):
                    print("Preliminary results:", e["preliminary_results"])

```

```go title="Go" lines
type toolCall struct {
	CallID    string `json:"call_id"`
	Name      string `json:"name"`
	Arguments string `json:"arguments"`
}

stream := func(input []any) []toolCall {
	body, _ := json.Marshal(map[string]any{
		"model":  "openai/gpt-5-nano",
		"input":  input,
		"tools":  []any{searchTool},
		"stream": true,
	})

	req, _ := http.NewRequest("POST", "https://openrouter.ai/api/v1/responses", bytes.NewReader(body))
	req.Header.Set("Authorization", "Bearer "+os.Getenv("OPENROUTER_API_KEY"))
	req.Header.Set("Content-Type", "application/json")

	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		panic(err)
	}
	defer resp.Body.Close()
	if resp.StatusCode != http.StatusOK {
		b, _ := io.ReadAll(resp.Body)
		panic(fmt.Sprintf("%s: %s", resp.Status, b))
	}

	var calls []toolCall
	scanner := bufio.NewScanner(resp.Body)
	scanner.Buffer(make([]byte, 1024*1024), 1024*1024)
	for scanner.Scan() {
		data, ok := strings.CutPrefix(scanner.Text(), "data: ")
		if !ok || data == "[DONE]" {
			continue
		}
		var event struct {
			Type  string `json:"type"`
			Delta string `json:"delta"`
			Item  struct {
				Type string `json:"type"`
				toolCall
			} `json:"item"`
		}
		if json.Unmarshal([]byte(data), &event) != nil {
			continue
		}
		switch event.Type {
		case "response.output_text.delta":
			fmt.Print(event.Delta)
		case "response.function_call_arguments.delta":
			fmt.Println("Tool argument delta:", event.Delta)
		case "response.completed":
			fmt.Println("Response complete")
		case "response.output_item.done":
			if event.Item.Type == "function_call" {
				calls = append(calls, event.Item.toolCall)
			}
		}
	}
	return calls
}

input := []any{map[string]any{"role": "user", "content": "Search for documents about TypeScript generics"}}
for calls := stream(input); len(calls) > 0; calls = stream(input) {
	for _, call := range calls {
		var preliminary []any
		// Run the generator tool, reporting intermediate progress
		result := searchDocs(call.Arguments, func(progress any) {
			fmt.Println("Progress:", progress)
			preliminary = append(preliminary, progress)
		})
		fmt.Println("Tool completed:", call.CallID)
		fmt.Println("Result:", result)
		if len(preliminary) > 0 {
			fmt.Println("Preliminary results:", preliminary)
		}
		output, _ := json.Marshal(result)
		input = append(input,
			map[string]any{"type": "function_call", "call_id": call.CallID, "name": call.Name, "arguments": call.Arguments},
			map[string]any{"type": "function_call_output", "call_id": call.CallID, "output": string(output)},
		)
	}
}
```

### Event Types

The full stream includes these event types:

| Event Type                               | Description                                          |
| ---------------------------------------- | ---------------------------------------------------- |
| `response.created`                       | Response object created                              |
| `response.in_progress`                   | Generation started                                   |
| `response.output_text.delta`             | Text content chunk                                   |
| `response.output_text.done`              | Text content complete                                |
| `response.reasoning.delta`               | Reasoning content chunk                              |
| `response.reasoning.done`                | Reasoning complete                                   |
| `response.function_call_arguments.delta` | Tool call argument chunk                             |
| `response.function_call_arguments.done`  | Tool call arguments complete                         |
| `response.completed`                     | Full response complete                               |
| `tool.preliminary_result`                | Progress from generator tools (intermediate yields)  |
| `tool.result`                            | Final result from tool execution                     |

```go title="Go" lines
			respCh <- e

	var text string
	var usage json.RawMessage

	wg.Add(2)

		defer wg.Done()
		for e := range textCh {
			if e.Type == "response.output_text.delta" {
				text += e.Delta
			}
		}
	}()

	// Consumer 2: Get full response
	go func() {
		defer wg.Done()
		for e := range respCh {
			if e.Type == "response.completed" {
				usage = e.Response.Usage
			}
		}
	}()
	wg.Wait()

	var pretty bytes.Buffer
	json.Indent(&pretty, usage, "", "  ")
```

```go title="Go" lines
// The same pattern a React component uses: append each delta to UI state,
// flip an isStreaming flag when done, and cancel the stream on cleanup.
// In React, call `onText` from setText, `onDone` from setIsStreaming(false),
// and return `cancel` from your useEffect cleanup.
// Failures reach the caller through `onError` instead of ending the process.
func streamChatResponse(prompt string, onText func(string), onDone func(), onError func(error)) (<-chan struct{}, context.CancelFunc) {
	ctx, cancel := context.WithCancel(context.Background())
	done := make(chan struct{})

	body, _ := json.Marshal(map[string]any{
		"model":  "openai/gpt-5-nano",
		"input":  prompt,
		"stream": true,
	})
	req, _ := http.NewRequestWithContext(ctx, "POST", "https://openrouter.ai/api/v1/responses", bytes.NewReader(body))
	req.Header.Set("Authorization", "Bearer "+os.Getenv("OPENROUTER_API_KEY"))
	req.Header.Set("Content-Type", "application/json")

	go func() {
		defer close(done)
		resp, err := http.DefaultClient.Do(req)
		if err != nil {
			if ctx.Err() != nil {
				return
			}
			onError(err)
			return
		}
		defer resp.Body.Close()
		if resp.StatusCode != http.StatusOK {
			b, _ := io.ReadAll(resp.Body)
			onError(fmt.Errorf("%s: %s", resp.Status, b))
			return
		}

		text := ""
		scanner := bufio.NewScanner(resp.Body)
		scanner.Buffer(make([]byte, 1024*1024), 1024*1024)
		for scanner.Scan() {
			data, ok := strings.CutPrefix(scanner.Text(), "data: ")
			if !ok || data == "[DONE]" {
				continue
			}
			var e struct {
				Type  string `json:"type"`
				Delta string `json:"delta"`
			}
			if json.Unmarshal([]byte(data), &e) != nil {
				continue
			}
			if e.Type == "response.failed" || e.Type == "error" {
				onError(fmt.Errorf("stream failed: %s", data))
				return
			}
			if e.Type == "response.output_text.delta" {
				text += e.Delta
				onText(text)
			}
		}
		// A cancelled or failed read ends the loop early: that is not a completed stream
		if ctx.Err() != nil {
			return
		}
		if err := scanner.Err(); err != nil {
			onError(err)
			return
		}
		onDone()
	}()

	return done, cancel
}

func main() {
```

```go title="Go" lines
		return
	}
	defer resp.Body.Close()
	if resp.StatusCode != http.StatusOK {
		b, _ := io.ReadAll(resp.Body)
		http.Error(w, resp.Status+": "+string(b), http.StatusBadGateway)
		return
	}

	w.Header().Set("Content-Type", "text/event-stream")
	w.Header().Set("Cache-Control", "no-cache")
	flusher := w.(http.Flusher)
	writeSSE := func(event string, data any) {
		b, _ := json.Marshal(data)
		fmt.Fprintf(w, "event: %s\ndata: %s\n\n", event, b)
		flusher.Flush()
	}

	scanner := bufio.NewScanner(resp.Body)
	scanner.Buffer(make([]byte, 1024*1024), 1024*1024)
	for scanner.Scan() {
		data, ok := strings.CutPrefix(scanner.Text(), "data: ")
		if !ok || data == "[DONE]" {
			continue
		}
		var e struct {
			Type  string `json:"type"`
			Delta string `json:"delta"`
		}
		if json.Unmarshal([]byte(data), &e) != nil {
			continue
		}
		if e.Type == "response.failed" || e.Type == "error" {
			writeSSE("error", json.RawMessage(data))
			return
		}
		if e.Type == "response.output_text.delta" {
			writeSSE("delta", map[string]string{"delta": e.Delta})
		}
	}
	// A read error means the upstream stream was cut short: report it instead of done
	if err := scanner.Err(); err != nil {
		writeSSE("error", map[string]string{"message": err.Error()})
		return
	}

	writeSSE("done", map[string]bool{"done": true})
}

func main() {
	mux := http.NewServeMux()
	mux.HandleFunc("/stream", streamHandler)

	// Try the endpoint in-process and print the SSE events as they arrive
	srv := httptest.NewServer(mux)
	defer srv.Close()

	resp, err := http.Get(srv.URL + "/stream?prompt=" + url.QueryEscape("Write a short haiku about streaming text."))
	if err != nil {
		panic(err)
	}
	defer resp.Body.Close()
	if resp.StatusCode != http.StatusOK {
		b, _ := io.ReadAll(resp.Body)
		panic(fmt.Sprintf("%s: %s", resp.Status, b))
	}
	io.Copy(os.Stdout, resp.Body)
}
```
