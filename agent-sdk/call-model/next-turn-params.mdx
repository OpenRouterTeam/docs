## Why nextTurnParams?

Traditional tool execution returns results to the model, but sometimes you need more:

- **Skills/Plugins**: Load domain-specific instructions when a skill is activated
- **Progressive Context**: Build up context as tools are used
- **Adaptive Behavior**: Adjust model parameters based on tool results
- **Clean Separation**: Tools manage their own context requirements

With `nextTurnParams`, tools can modify any `callModel` parameter for the next turn.

## Basic Example

<CodeGroup>

```typescript expandable lines
import { tool } from '@openrouter/agent';
import { z } from 'zod';

const expertModeTool = tool({
  name: 'enable_expert_mode',
  description: 'Enable expert mode for detailed technical responses',
  inputSchema: z.object({
    domain: z.string().describe('Technical domain (e.g., "kubernetes", "react")'),
  }),
  outputSchema: z.object({ enabled: z.boolean() }),

  nextTurnParams: {
    instructions: (params, context) => {
      const base = context.instructions ?? '';
      return `${base}

EXPERT MODE ENABLED for ${params.domain}:
- Provide detailed technical explanations
- Include code examples and best practices
- Reference official documentation
- Assume advanced knowledge`;
    },
    temperature: () => 0.3, // More precise for technical content
  },

  execute: async (params) => {
    return { enabled: true };
  },
});
```

```python title="Python" lines
from pydantic import BaseModel, Field
from openrouter_agent import tool

class ExpertModeInput(BaseModel):
    domain: str = Field(description='Technical domain (e.g., "kubernetes", "react")')

class ExpertModeOutput(BaseModel):
    enabled: bool

def expert_instructions(params, context):
    base = context.get("instructions") or ""
    return f"""{base}

EXPERT MODE ENABLED for {params['domain']}:
- Provide detailed technical explanations
- Include code examples and best practices
- Reference official documentation
- Assume advanced knowledge"""

expert_mode_tool = tool(
    name="enable_expert_mode",
    description="Enable expert mode for detailed technical responses",
    input_schema=ExpertModeInput,
    output_schema=ExpertModeOutput,
    next_turn_params={
        "instructions": expert_instructions,
        "temperature": lambda params, context: 0.3,  # More precise for technical content
    },
    execute=lambda params, ctx: ExpertModeOutput(enabled=True),
)

```

```go title="Go" lines
// Go has no agent SDK, so the tool and its nextTurnParams are plain values
// that your tool-calling loop applies after the tool runs.
type NextTurnParams struct {
	Instructions func(args map[string]any, base string) string
	Temperature  func() float64
}

type Tool struct {
	Name           string
	Description    string
	Parameters     map[string]any
	NextTurnParams NextTurnParams
	Execute        func(args map[string]any) map[string]any
}

var expertModeTool = Tool{
	Name:        "enable_expert_mode",
	Description: "Enable expert mode for detailed technical responses",

		"type": "object",

			"domain": map[string]any{"type": "string", "description": `Technical domain (e.g., "kubernetes", "react")`},
		},
		"required": []string{"domain"},
	},

	NextTurnParams: NextTurnParams{
		Instructions: func(args map[string]any, base string) string {
			return fmt.Sprintf(`%s

- Provide detailed technical explanations
- Include code examples and best practices
- Reference official documentation
- Assume advanced knowledge`, base, args["domain"])
		},
		Temperature: func() float64 { return 0.3 }, // More precise for technical content
	},

	Execute: func(args map[string]any) map[string]any {
		return map[string]any{"enabled": true}
	},
}
```

## The Claude Code Skills Pattern

This example shows how to recreate Claude Code's skills system as a single encapsulated tool:

<CodeGroup>

```typescript expandable lines
import { OpenRouter } from '@openrouter/sdk';
import { callModel, tool } from '@openrouter/agent';
import { existsSync, readFileSync } from 'fs';
import { homedir } from 'os';
import { join } from 'path';
import { z } from 'zod';

const openrouter = new OpenRouter({ apiKey: process.env.OPENROUTER_API_KEY });

const skillsTool = tool({
  name: "skill",
  description: `Load a specialized skill to enhance the assistant's capabilities.
Available skills: pdf-processing, data-analysis, code-review, etc.
Each skill provides domain-specific instructions and capabilities.`,
  inputSchema: z.object({
    type: z.string().describe("The skill type to load (e.g., 'pdf-processing')"),
  }),
  outputSchema: z.string(),

  // nextTurnParams runs after all tool calls execute, before responses go to model
  // Executed in order of tools array. This is where the magic happens.
  nextTurnParams: {
    input: (params, context) => {
      // Prevent duplicate skill loading
      if (JSON.stringify(context.input).includes(`Skill ${params.type} is already loaded`)) {
        return context.input;
      }

      // The skill name comes from the model, so accept a plain directory name only
      if (!/^[A-Za-z0-9][A-Za-z0-9_-]*$/.test(params.type)) {
        return context.input;
      }

      // Load the skill's instructions from file system
      const skillDir = join(homedir(), '.claude', 'skills', params.type);
      const skillPath = join(skillDir, 'SKILL.md');
      if (!existsSync(skillPath)) {
        return context.input;
      }
      const skill = readFileSync(skillPath, "utf-8");

      const history = Array.isArray(context.input)
        ? context.input
        : [{ role: "user", content: context.input }];

      // Inject skill context into the conversation
      return [
        ...history,
        {
          role: "user",
          content: `Base directory for this skill: ${skillDir}/

${skill}`,
        },
      ];
    },
  },

  execute: async (params, context) => {
    // Check if already loaded
    if (JSON.stringify(context?.input ?? []).includes(`Skill ${params.type} is already loaded`)) {
      return `Skill ${params.type} is already loaded`;
    }

    return `Launching skill ${params.type}`;
  },
});

// Usage - the skill automatically enriches future turns
const result = callModel(openrouter, {
  model: 'anthropic/claude-sonnet-4.5',
  input: 'Process this PDF and extract the key findings',
  tools: [skillsTool] as const,
});

console.log(await result.getText());

```

```python title="Python" lines
import asyncio
import json
import os
import re
from pathlib import Path
from pydantic import BaseModel, Field
from openrouter_agent import OpenRouter, call_model, tool

class SkillInput(BaseModel):
    type: str = Field(description="The skill type to load (e.g., 'pdf-processing')")

# next_turn_params runs after all tool calls execute, before responses go to model
# Executed in order of tools array. This is where the magic happens.
def skill_input(params, context):
    current = context.get("input")
    # Prevent duplicate skill loading: the marker is written below, with the instructions
    if f"Skill {params['type']} is loaded" in json.dumps(current, default=str):
        return current

    # The skill name comes from the model, so accept a plain directory name only
    if not re.fullmatch(r"[A-Za-z0-9][A-Za-z0-9_-]*", params["type"]):
        return current

    # Load the skill's instructions from file system
    skill_dir = Path.home() / ".claude" / "skills" / params["type"]
    skill_path = skill_dir / "SKILL.md"
    if not skill_path.exists():
        return current
    skill = skill_path.read_text(encoding="utf-8")

    history = current if isinstance(current, list) else [{"role": "user", "content": current}]

    # Inject skill context into the conversation
    return [
        *history,
        {"role": "user", "content": f"Skill {params['type']} is loaded. Base directory for this skill: {skill_dir}/\n\n{skill}"},
    ]

def execute_skill(params, ctx):
    # Check if already loaded
    current = (ctx or {}).get("input", [])
    if f"Skill {params.type} is loaded" in json.dumps(current, default=str):
        return f"Skill {params.type} is already loaded"

    # Report a load error instead of a launch when there are no instructions to add
    skill_path = Path.home() / ".claude" / "skills" / params.type / "SKILL.md"
    if not re.fullmatch(r"[A-Za-z0-9][A-Za-z0-9_-]*", params.type) or not skill_path.exists():
        return f"Skill {params.type} could not be loaded: no SKILL.md found for that name"
    return f"Launching skill {params.type}"

skills_tool = tool(
    name="skill",
    description="""Load a specialized skill to enhance the assistant's capabilities.
Available skills: pdf-processing, data-analysis, code-review, etc.
Each skill provides domain-specific instructions and capabilities.""",
    input_schema=SkillInput,
    next_turn_params={"input": skill_input},
    execute=execute_skill,
)

async def main():
    client = OpenRouter(api_key=os.environ["OPENROUTER_API_KEY"])

    # Usage - the skill automatically enriches future turns
    result = call_model(client, {
        "model": "anthropic/claude-sonnet-4.5",
        "input": "Process this PDF and extract the key findings",
        "tools": [skills_tool],
    })

    print(await result.get_text())

asyncio.run(main())

```

```go title="Go" lines
package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"os"
	"path/filepath"
	"regexp"
	"strings"
)

// The skill name comes from the model, so accept a plain directory name only
var skillName = regexp.MustCompile(`^[A-Za-z0-9][A-Za-z0-9_-]*$`)

type toolCall struct {
	ID       string `json:"id"`
	Type     string `json:"type"`
	Function struct {
		Name      string `json:"name"`
		Arguments string `json:"arguments"`
	} `json:"function"`
}

type message struct {
	Role       string     `json:"role"`
	Content    any        `json:"content"`
	ToolCalls  []toolCall `json:"tool_calls,omitempty"`
	ToolCallID string     `json:"tool_call_id,omitempty"`
}

var skillTool = map[string]any{
	"type": "function",
	"function": map[string]any{
		"name": "skill",
		"description": `Load a specialized skill to enhance the assistant's capabilities.
Available skills: pdf-processing, data-analysis, code-review, etc.
Each skill provides domain-specific instructions and capabilities.`,
		"parameters": map[string]any{
			"type": "object",
			"properties": map[string]any{
				"type": map[string]any{"type": "string", "description": "The skill type to load (e.g., 'pdf-processing')"},
			},
			"required": []string{"type"},
		},
	},
}

// alreadyLoaded looks for the marker nextTurnInput writes with the instructions
func alreadyLoaded(history []message, skill string) bool {
	b, _ := json.Marshal(history)
	return strings.Contains(string(b), fmt.Sprintf("Skill %s is loaded", skill))
}

func skillDir(skill string) string {
	home, _ := os.UserHomeDir()
	return filepath.Join(home, ".claude", "skills", skill)
}

// execute: report whether the skill is launching, already loaded, or cannot be loaded
func execute(skill string, history []message) string {
	if alreadyLoaded(history, skill) {
		return fmt.Sprintf("Skill %s is already loaded", skill)
	}
	if !skillName.MatchString(skill) {
		return fmt.Sprintf("Skill %s could not be loaded: no SKILL.md found for that name", skill)
	}
	if _, err := os.Stat(filepath.Join(skillDir(skill), "SKILL.md")); err != nil {
		return fmt.Sprintf("Skill %s could not be loaded: no SKILL.md found for that name", skill)
	}
	return fmt.Sprintf("Launching skill %s", skill)
}

// nextTurnParams.input: runs after tool calls execute, before the next model turn
func nextTurnInput(skill string, history []message) []message {
	if alreadyLoaded(history, skill) || !skillName.MatchString(skill) {
		return history
	}
	dir := skillDir(skill)
	content, err := os.ReadFile(filepath.Join(dir, "SKILL.md"))
	if err != nil {
		return history
	}
	return append(history, message{Role: "user", Content: fmt.Sprintf("Skill %s is loaded. Base directory for this skill: %s/\n\n%s", skill, dir, content)})
}

func main() {
	history := []message{{Role: "user", Content: "Process this PDF and extract the key findings"}}

	for turn := 0; turn < 5; turn++ {
		body, _ := json.Marshal(map[string]any{
			"model":    "anthropic/claude-sonnet-4.5",
			"messages": history,
			"tools":    []any{skillTool},
		})
		req, _ := http.NewRequest("POST", "https://openrouter.ai/api/v1/chat/completions", bytes.NewReader(body))
		req.Header.Set("Authorization", "Bearer "+os.Getenv("OPENROUTER_API_KEY"))
		req.Header.Set("Content-Type", "application/json")
		resp, err := http.DefaultClient.Do(req)
		if err != nil {
			panic(err)
		}
		var out struct {
			Choices []struct {
				Message message `json:"message"`
			} `json:"choices"`
		}
		raw, err := io.ReadAll(resp.Body)
		resp.Body.Close()
		if err != nil {
			panic(err)
		}
		if resp.StatusCode != http.StatusOK {
			panic(fmt.Sprintf("request failed: %s: %s", resp.Status, raw))
		}
		if err := json.Unmarshal(raw, &out); err != nil {
			panic(fmt.Sprintf("decode response: %v", err))
		}
		if len(out.Choices) == 0 {
			panic(fmt.Sprintf("no choices in response: %s", raw))
		}
		msg := out.Choices[0].Message
		if len(msg.ToolCalls) == 0 {
			fmt.Println(msg.Content)
			return
		}

		history = append(history, msg)
		var skills []string
		for _, tc := range msg.ToolCalls {
			var args struct {
				Type string `json:"type"`
			}
			// Reject malformed arguments or a missing type before running the skill
			if err := json.Unmarshal([]byte(tc.Function.Arguments), &args); err != nil || args.Type == "" {
				history = append(history, message{Role: "tool", ToolCallID: tc.ID, Content: "Invalid skill arguments: a string \"type\" is required"})
				continue
			}
			history = append(history, message{Role: "tool", ToolCallID: tc.ID, Content: execute(args.Type, history)})
			skills = append(skills, args.Type)
		}
		for _, s := range skills {
			history = nextTurnInput(s, history)
		}
	}

	// The last reply still asked for a tool, so there is no answer to print
	fmt.Println("Stopped after 5 turns without a final answer")
}
```

### Key Benefits

1. **Encapsulation**: Skill loading logic is entirely contained in the tool
2. **Idempotency**: Built-in check prevents loading the same skill twice
3. **Clean API**: Callers don't need to know about skill file locations
4. **Composability**: Multiple skills can be loaded across turns

## Execution Order

Understanding when `nextTurnParams` runs is crucial:

```lines
1. Model generates tool calls
   ↓
2. All tool `execute` functions run
   ↓
3. `nextTurnParams` functions run for each tool (in tools array order)
   ↓
4. Modified parameters used for next model turn
   ↓
5. Repeat until model stops calling tools
```

## Available Context

`nextTurnParams` functions receive two arguments:

### params

The validated input parameters that were passed to the tool:

### context

The current request context, including:

| Property | Type | Description |
|----------|------|-------------|
| `input` | `OpenResponsesInput` | Current message history |
| `model` | `string \| undefined` | Current model selection |
| `models` | `string[] \| undefined` | Model fallback array |
| `instructions` | `string \| undefined` | Current system instructions |
| `temperature` | `number \| undefined` | Current temperature |
| `maxOutputTokens` | `number \| undefined` | Current max tokens |
| `topP` | `number \| undefined` | Current top-p sampling |
| `topK` | `number \| undefined` | Current top-k sampling |

```go title="Go" lines
// nextTurnParams: each field runs after tool calls execute, before the next model turn
type turnParams struct {
	Input           []message
	Model           string
	Instructions    string
	Temperature     float64
	MaxOutputTokens int
}

func nextTurnParams(params skillInput, ctx turnParams) turnParams {
	// Modify message history
	ctx.Input = append(ctx.Input, newMessage)

	// Change model
	ctx.Model = "anthropic/claude-sonnet-4.5"
```

### Complexity-Based Model Selection

Upgrade to better models when needed. In the Python tab, `analyze_code_complexity` is application-defined: a synchronous function that takes the code and returns a `ComplexityOutput`. The callbacks receive the tool input, not its output, so they call it again; the Go tab runs its own loop and passes the tool result in directly.

<CodeGroup>

```typescript expandable lines
const complexityAnalyzer = tool({
  name: "analyze_complexity",
  inputSchema: z.object({ code: z.string() }),
  outputSchema: z.object({ complexity: z.enum(['low', 'medium', 'high']) }),

  nextTurnParams: {
    model: (params, context) => {
      // Upgrade to more capable model for complex code
      if (params.complexity === 'high') {
        return 'anthropic/claude-sonnet-4.5';
      }
      return context.model ?? 'openai/gpt-5-nano';
    },
    temperature: (params, context) => {
      // Lower temperature for complex analysis
      return params.complexity === 'high' ? 0.3 : 0.7;
    },
  },

  execute: async (params) => {
    return analyzeCodeComplexity(params.code);
  },
});
```

```python title="Python" lines
class ComplexityInput(BaseModel):
    code: str

class ComplexityOutput(BaseModel):
    complexity: Literal["low", "medium", "high"]

# params is the tool input ({"code": ...}), so derive the complexity from it
def complexity_of(params):
    return analyze_code_complexity(params["code"]).complexity

def select_model(params, context):
    # Upgrade to more capable model for complex code
    if complexity_of(params) == "high":
        return "anthropic/claude-sonnet-4.5"
    return context.get("model") or "openai/gpt-5-nano"

def select_temperature(params, context):
    # Lower temperature for complex analysis
    return 0.3 if complexity_of(params) == "high" else 0.7

async def run_analysis(params, ctx):
    return analyze_code_complexity(params.code)

complexity_analyzer = tool(
    name="analyze_complexity",
    input_schema=ComplexityInput,
    output_schema=ComplexityOutput,
    next_turn_params={
        "model": select_model,
        "temperature": select_temperature,
    },
    execute=run_analysis,
)
```

```typescript expandable lines
    failed: z.array(z.object({ name: z.string(), reason: z.string() })),
  }),

  nextTurnParams: {
    input: (params, context) => {
      let newInput = context.input;

      for (const skillName of params.skills) {
        const skillPath = `~/.skills/${skillName}/SKILL.md`;
        if (!existsSync(skillPath)) continue;

        const skillMarker = `[Skill: ${skillName}]`;
        if (JSON.stringify(newInput).includes(skillMarker)) continue;

        const skillContent = readFileSync(skillPath, 'utf-8');
        newInput = [
          ...(Array.isArray(newInput) ? newInput : [newInput]),
          { role: 'user', content: `${skillMarker}\n${skillContent}` },
        ];
      }

      return newInput;
    },
  },

  execute: async ({ skills }) => {
    const loaded = [];
    const failed = [];

    for (const skill of skills) {
      if (existsSync(`~/.skills/${skill}/SKILL.md`)) {
        loaded.push(skill);
      } else {
        failed.push({ name: skill, reason: 'Not found' });
      }
    }

    return { loaded, failed };
  },
});
```

```go title="Go" lines
var skillName = regexp.MustCompile(`^[A-Za-z0-9][A-Za-z0-9_-]*$`)

func skillPath(name string) (string, bool) {
	if !skillName.MatchString(name) {
		return "", false
	}
	home, err := os.UserHomeDir()
	if err != nil {
		return "", false
	}
	return filepath.Join(home, ".skills", name, "SKILL.md"), true
}

// nextTurnParams.input: append each new skill's content to the conversation
func loadSkillsInput(params skillsInput, input []map[string]any) []map[string]any {
	newInput := append([]map[string]any{}, input...)

	for _, name := range params.Skills {
		path, ok := skillPath(name)
		if !ok {
			continue
		}
		content, err := os.ReadFile(path)
		if err != nil {
			continue
		}

		marker := fmt.Sprintf("[Skill: %s]", name)
		existing, _ := json.Marshal(newInput)
		if strings.Contains(string(existing), marker) {
			continue
		}

		newInput = append(newInput, map[string]any{
			"role":    "user",
			"content": marker + "\n" + string(content),
		})
	}

	return newInput
}

func executeLoadSkills(params skillsInput) skillsOutput {
	out := skillsOutput{Loaded: []string{}, Failed: []failedSkill{}}

	for _, skill := range params.Skills {
		path, ok := skillPath(skill)
		if _, err := os.Stat(path); ok && err == nil {
			out.Loaded = append(out.Loaded, skill)
		} else {
			out.Failed = append(out.Failed, failedSkill{Name: skill, Reason: "Not found"})
		}
	}

	return out
}
```

### Minimal Modifications

Only modify what's necessary:

<CodeGroup>

```typescript lines
// Good: Minimal, targeted change
nextTurnParams: {
  temperature: (params) => params.needsPrecision ? 0.2 : undefined,
},

// Avoid: Unnecessary spreading
nextTurnParams: {
  temperature: (params, ctx) => {
    return params.needsPrecision ? 0.2 : ctx.temperature;
```

```python title="Python" lines
next_turn_params = {
```
