## Basic Usage

Any parameter in `callModel` can be a function that computes its value based on conversation context. This enables adaptive behavior - changing models, adjusting temperature, or modifying instructions as the conversation evolves.

Pass a function instead of a static value:

<CodeGroup>

```typescript lines
import { OpenRouter, callModel, tool } from '@openrouter/agent';
import { z } from 'zod';

const openrouter = new OpenRouter({
  apiKey: process.env.OPENROUTER_API_KEY,
});

const myTool = tool({
  name: 'get_time',
  description: 'Get the current time',
  inputSchema: z.object({}),
  execute: async () => ({ time: new Date().toISOString() }),
});

const result = callModel(openrouter, {
  // Dynamic model selection based on turn count
  model: (ctx) => {
    return ctx.numberOfTurns > 3 ? 'openai/gpt-5.2' : 'openai/gpt-5-nano';
  },
  input: 'Hello!',
  tools: [myTool] as const,
});

const text = await result.getText();
console.log(text);

```

```python title="Python" lines
import asyncio
import os
from datetime import datetime, timezone
from pydantic import BaseModel
from openrouter_agent import OpenRouter, call_model, tool


class TimeInput(BaseModel):
    pass


class TimeOutput(BaseModel):
    time: str


async def main() -> None:
    client = OpenRouter(api_key=os.environ["OPENROUTER_API_KEY"])

    my_tool = tool(
        name="get_time",
        description="Get the current time",
        input_schema=TimeInput,
        output_schema=TimeOutput,
        execute=lambda params, ctx: TimeOutput(
            time=datetime.now(timezone.utc).isoformat()
        ),
    )

    result = call_model(
        client,
        {
            # Dynamic model selection based on turn count
            "model": lambda ctx: (
                "openai/gpt-5.2" if ctx["number_of_turns"] > 3 else "openai/gpt-5-nano"
            ),
            "input": "Hello!",
            "tools": [my_tool],
        },
    )

    text = await result.get_text()
    print(text)


asyncio.run(main())

```

```go title="Go" lines
package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	"io"
	"log"
	"net/http"
	"os"
	"time"
)

type obj = map[string]any

// Dynamic model selection based on turn count
func selectModel(numberOfTurns int) string {
	if numberOfTurns > 3 {
		return "openai/gpt-5.2"
	}
	return "openai/gpt-5-nano"
}

func main() {
	tools := []any{obj{"type": "function", "function": obj{
		"name":        "get_time",
		"description": "Get the current time",
		"parameters":  obj{"type": "object", "properties": obj{}},
	}}}
	messages := []any{obj{"role": "user", "content": "Hello!"}}

	for turns := 1; ; turns++ {
		body, _ := json.Marshal(obj{"model": selectModel(turns), "messages": messages, "tools": tools})
		req, _ := http.NewRequest("POST", "https://openrouter.ai/api/v1/chat/completions", bytes.NewReader(body))
		req.Header.Set("Authorization", "Bearer "+os.Getenv("OPENROUTER_API_KEY"))
		req.Header.Set("Content-Type", "application/json")
		resp, err := http.DefaultClient.Do(req)
		if err != nil {
			log.Fatal(err)
		}
		raw, _ := io.ReadAll(resp.Body)
		resp.Body.Close()
		if resp.StatusCode != http.StatusOK {
			log.Fatalf("%s: %s", resp.Status, raw)
		}

		var out struct {
			Choices []struct {
				Message json.RawMessage `json:"message"`
			} `json:"choices"`
		}
		if err := json.Unmarshal(raw, &out); err != nil || len(out.Choices) == 0 {
			log.Fatalf("unexpected response: %s", raw)
		}
		var msg struct {
			Content   string `json:"content"`
			ToolCalls []struct {
				ID string `json:"id"`
			} `json:"tool_calls"`
		}
		json.Unmarshal(out.Choices[0].Message, &msg)

		if len(msg.ToolCalls) == 0 {
			fmt.Println(msg.Content)
			return
		}

		messages = append(messages, out.Choices[0].Message)
		for _, tc := range msg.ToolCalls {
			result, _ := json.Marshal(obj{"time": time.Now().UTC().Format(time.RFC3339)})
			messages = append(messages, obj{"role": "tool", "tool_call_id": tc.ID, "content": string(result)})
		}
	}
}
```

The Go tab above is a complete program. The later Go tabs on this page are fragments that go inside `func main()` and call the `sendChat` helper below. The helper is a top-level declaration that reuses the imports and the `obj` alias of the program above. It posts one chat completions request, checks the HTTP status, and returns the assistant message. `Raw` holds the message as returned, ready to append to `messages`, and each tool call exposes its `ID` and `Function.Arguments`. Names such as `fetchUserPreferences`, `analysisTool`, and `runAnalysis` are application-defined: supply your own.

```go title="Go" lines
type chatMessage struct {
	Content   string `json:"content"`
	ToolCalls []struct {
		ID       string `json:"id"`
		Function struct {
			Arguments string `json:"arguments"`
		} `json:"function"`
	} `json:"tool_calls"`
	Raw json.RawMessage `json:"-"`
}

func sendChat(payload obj) (*chatMessage, error) {
	body, _ := json.Marshal(payload)
	req, _ := http.NewRequest("POST", "https://openrouter.ai/api/v1/chat/completions", bytes.NewReader(body))
	req.Header.Set("Authorization", "Bearer "+os.Getenv("OPENROUTER_API_KEY"))
	req.Header.Set("Content-Type", "application/json")
	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		return nil, err
	}
	defer resp.Body.Close()
	raw, err := io.ReadAll(resp.Body)
	if err != nil {
		return nil, err
	}
	if resp.StatusCode != http.StatusOK {
		return nil, fmt.Errorf("%s: %s", resp.Status, raw)
	}
	var out struct {
		Choices []struct {
			Message json.RawMessage `json:"message"`
		} `json:"choices"`
	}
	if err := json.Unmarshal(raw, &out); err != nil || len(out.Choices) == 0 {
		return nil, fmt.Errorf("unexpected response: %s", raw)
	}
	msg := &chatMessage{Raw: out.Choices[0].Message}
	if err := json.Unmarshal(out.Choices[0].Message, msg); err != nil {
		return nil, err
	}
	return msg, nil
}
```

## Function Signature

Parameter functions receive a `TurnContext` and return the parameter value:

```typescript lines
type ParameterFunction<T> = (context: TurnContext) => T | Promise<T>;
```

### TurnContext

| Property | Type | Description |
|----------|------|-------------|
| `numberOfTurns` | `number` | Current turn number (1-indexed) |
| `turnRequest` | `OpenResponsesRequest \| undefined` | Current request object containing messages and model settings |
| `toolCall` | `OpenResponsesFunctionToolCall \| undefined` | The specific tool call being executed |

## Async Functions

Functions can be async for fetching external data:

<CodeGroup>

## Common Patterns

### Progressive Model Upgrade

Start with a fast model, upgrade for complex tasks:

<CodeGroup>

```typescript lines
const result = callModel(openrouter, {
  model: (ctx) => {
    // First few turns: fast model
    if (ctx.numberOfTurns <= 2) {
      return 'openai/gpt-5-nano';
    }

    // Complex conversations: capable model
    return 'openai/gpt-5.2';
  },
  input: 'Let me think through this problem...',
  tools: [analysisTool] as const,
});
```

```python title="Python" lines
def select_model(ctx):
```

### Adaptive Temperature

Adjust creativity based on context:

<CodeGroup>

```typescript lines
const input = 'Write a creative story';

const result = callModel(openrouter, {
  model: 'openai/gpt-5-nano',
  temperature: (ctx) => {
    // Analyze recent messages for task type
    // (turnRequest may be undefined on the first turn: fall back to the initial input)
    const lastMessage = JSON.stringify(ctx.turnRequest?.input ?? input).toLowerCase();

    if (lastMessage.includes('creative') || lastMessage.includes('brainstorm')) {
      return 1.0; // Creative tasks
    }
    if (lastMessage.includes('code') || lastMessage.includes('calculate')) {
      return 0.2; // Precise tasks
    }
    return 0.7; // Default
  },
  input,
});
```

```python title="Python" lines
initial_input = "Write a creative story"


def temperature(ctx):
    # Analyze recent messages for task type

    if "creative" in last_message or "brainstorm" in last_message:
        return 1.0  # Creative tasks
    if "code" in last_message or "calculate" in last_message:
        return 0.2  # Precise tasks
    return 0.7  # Default


result = call_model(
    client,
    {
        "model": "openai/gpt-5-nano",

    },
)

```

### Context-Aware Instructions

Build instructions based on conversation state:

<CodeGroup>

```typescript lines
const result = openrouter.callModel({
  model: 'openai/gpt-5-nano',
  instructions: (ctx) => {
    const base = 'You are a helpful assistant.';
    const turnInfo = `This is turn ${ctx.numberOfTurns} of the conversation.`;

    // Add context based on history length
    if (ctx.numberOfTurns > 5) {
      return `${base}\n${turnInfo}\nKeep responses concise - this is a long conversation.`;
    }

    return `${base}\n${turnInfo}`;
  },
  input: 'Continue helping me...',
  tools: [helpTool],
```

## Combining with Tools

Dynamic parameters work alongside tool execution:

<CodeGroup>

## Execution Order

Dynamic parameters are resolved at the start of each turn:

```lines
1. Resolve all parameter functions with current TurnContext
2. Build request with resolved values
3. Send to model
4. Execute tools (if any)
5. Check stop conditions
6. Update TurnContext for next turn
7. Repeat from step 1
```

## Best Practices

### Keep Functions Pure

Avoid side effects in parameter functions:

<CodeGroup>

```typescript lines
// Good: Pure function
model: (ctx) => ctx.numberOfTurns > 3 ? 'gpt-4' : 'gpt-4o-mini',

// Avoid: Side effects
model: (ctx) => {
  logToDatabase(ctx); // Side effect
  return 'gpt-4';
},
```

```python title="Python" lines
# Good: Pure function
def model(ctx):
    return "gpt-4" if ctx["number_of_turns"] > 3 else "gpt-4o-mini"


# Avoid: Side effects
def model(ctx):
    log_to_database(ctx)  # Side effect
    return "gpt-4"

```

```go title="Go" lines
// Good: Pure function
modelFor := func(numberOfTurns int) string {
	if numberOfTurns > 3 {
		return "openai/gpt-4"
	}
	return "openai/gpt-4o-mini"
}

// Avoid: Side effects
modelWithSideEffect := func(numberOfTurns int) string {
	logToDatabase(numberOfTurns) // Side effect
	return "openai/gpt-4"
}

```

</CodeGroup>

### Cache Expensive Operations

Cache results for repeated calls:

<CodeGroup>

```typescript lines
let cachedRules: string | null = null;

const result = callModel(openrouter, {
  model: 'openai/gpt-5-nano',
  instructions: async (ctx) => {
    if (!cachedRules) {
      cachedRules = await fetchExpensiveRules();
    }
    return cachedRules;
  },
  input: 'Hello!',
});
```

```python title="Python" lines
cached_rules = None


async def instructions(ctx):
    global cached_rules
    if cached_rules is None:
        cached_rules = await fetch_expensive_rules()
    return cached_rules


result = call_model(
    client,
    {
        "model": "openai/gpt-5-nano",
        "instructions": instructions,
        "input": "Hello!",
    },
)

```

```go title="Go" lines
var cachedRules string

instructions := func() string {
	if cachedRules == "" {
		cachedRules = fetchExpensiveRules()
	}
	return cachedRules
}

msg, err := sendChat(obj{
	"model": "openai/gpt-5-nano",
	"messages": []any{
		obj{"role": "system", "content": instructions()},
		obj{"role": "user", "content": "Hello!"},
	},
})
if err != nil {
	log.Fatal(err)
}
result := msg.Content
```

</CodeGroup>

### Use Sensible Defaults

Always have fallback values:

<CodeGroup>

```typescript lines
model: (ctx) => {
  const preferredModel = getPreferredModel();
  return preferredModel ?? 'openai/gpt-5-nano'; // Default fallback
},
```

```python title="Python" lines
def model(ctx):
    preferred_model = get_preferred_model()
    return preferred_model if preferred_model is not None else "openai/gpt-5-nano"  # Default fallback

```

```go title="Go" lines
modelFor := func() string {
	if preferredModel := getPreferredModel(); preferredModel != "" {
		return preferredModel
	}
	return "openai/gpt-5-nano" // Default fallback
}

```

</CodeGroup>
