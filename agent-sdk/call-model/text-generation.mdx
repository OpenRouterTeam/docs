## Basic Usage

The simplest way to generate text:

<CodeGroup>

```typescript lines
import { OpenRouter, callModel } from '@openrouter/agent';

const openrouter = new OpenRouter({
  apiKey: process.env.OPENROUTER_API_KEY,
});

const result = callModel(openrouter, {
  model: 'openai/gpt-5-nano',
  input: 'Explain quantum computing in one sentence.',
});

const text = await result.getText();
console.log(text);

```

```python title="Python" lines
import asyncio
import os

from openrouter_agent import OpenRouter, call_model


async def main() -> None:
    openrouter = OpenRouter(api_key=os.environ["OPENROUTER_API_KEY"])

    result = call_model(
        openrouter,
        {
            "model": "openai/gpt-5-nano",
            "input": "Explain quantum computing in one sentence.",
        },
    )

    text = await result.get_text()
    print(text)


asyncio.run(main())

```

```go title="Go" lines
package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	"log"
	"net/http"
	"os"
)

func main() {
	body, _ := json.Marshal(map[string]any{
		"model": "openai/gpt-5-nano",
		"input": "Explain quantum computing in one sentence.",
	})

	req, _ := http.NewRequest("POST", "https://openrouter.ai/api/v1/responses", bytes.NewReader(body))
	req.Header.Set("Authorization", "Bearer "+os.Getenv("OPENROUTER_API_KEY"))
	req.Header.Set("Content-Type", "application/json")

	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		log.Fatal(err)
	}
	defer resp.Body.Close()
	if resp.StatusCode != http.StatusOK {
		log.Fatalf("unexpected status: %s", resp.Status)
	}

	var result struct {
		Output []struct {
			Type    string `json:"type"`
			Content []struct {
				Type string `json:"type"`
				Text string `json:"text"`
			} `json:"content"`
		} `json:"output"`
	}
	if err := json.NewDecoder(resp.Body).Decode(&result); err != nil {
		log.Fatal(err)
	}

	text := ""
	for _, item := range result.Output {
		if item.Type != "message" {
			continue
		}
		for _, c := range item.Content {
			if c.Type == "output_text" {
				text += c.Text
			}
		}
	}
	fmt.Println(text)
}

```

```bash title="cURL" lines
curl -s https://openrouter.ai/api/v1/responses \
  -H "Authorization: Bearer $OPENROUTER_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "model": "openai/gpt-5-nano",
    "input": "Explain quantum computing in one sentence."
  }' | jq -r '.output[] | select(.type == "message") | .content[] | select(.type == "output_text") | .text'

```

</CodeGroup>


## Input Formats

callModel accepts several input formats to match your use case.

### String Input

<CodeGroup>

```typescript lines

  input: 'What is the speed of light?',
```

## Error Handling

Handle common error cases:

<CodeGroup>

```typescript lines
try {
  const result = callModel(openrouter, {
    model: 'openai/gpt-5-nano',
    input: 'Hello!',
  });

  const text = await result.getText();
  console.log(text);
} catch (error) {
  if (error instanceof Error && 'statusCode' in error) {
    if (error.statusCode === 401) {
      console.error('Invalid API key');
    } else if (error.statusCode === 429) {
      console.error('Rate limited - try again later');
    } else if (error.statusCode === 503) {
      console.error('Model unavailable');
    } else {
      console.error('Request failed with status', error.statusCode);
    }
  } else {
    console.error('Unexpected error:', error);
  }
}
```

```go title="Go" lines
package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	"log"
	"net/http"
	"os"
	"sync"
)

func getText(input string) (string, error) {
	body, _ := json.Marshal(map[string]any{"model": "openai/gpt-5-nano", "input": input})

	req, _ := http.NewRequest("POST", "https://openrouter.ai/api/v1/responses", bytes.NewReader(body))
	req.Header.Set("Authorization", "Bearer "+os.Getenv("OPENROUTER_API_KEY"))
	req.Header.Set("Content-Type", "application/json")

	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		return "", err
	}
	defer resp.Body.Close()
	if resp.StatusCode != http.StatusOK {
		return "", fmt.Errorf("unexpected status: %s", resp.Status)
	}

	var result struct {
		Output []struct {
			Type    string `json:"type"`
			Content []struct {
				Type string `json:"type"`
				Text string `json:"text"`
			} `json:"content"`
		} `json:"output"`
	}
	if err := json.NewDecoder(resp.Body).Decode(&result); err != nil {
		return "", err
	}

	text := ""
	for _, item := range result.Output {
		if item.Type != "message" {
			continue
		}
		for _, c := range item.Content {
			if c.Type == "output_text" {
				text += c.Text
			}
		}
	}
	return text, nil
}

func main() {
	inputs := []string{"Question 1", "Question 2", "Question 3"}
	results := make([]string, len(inputs))
	errs := make([]error, len(inputs))

	var wg sync.WaitGroup
	for i, input := range inputs {
		wg.Add(1)
		go func() {
			defer wg.Done()
			results[i], errs[i] = getText(input)
		}()
	}
	wg.Wait()

	for i, result := range results {
		if errs[i] != nil {
			log.Fatal(errs[i])
		}
		fmt.Printf("Result %d: %s\n", i+1, result)
	}
}
```
