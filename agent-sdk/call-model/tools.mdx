## The tool() Helper

The `tool()` function creates type-safe tools with Zod schema validation:

<CodeGroup>

```typescript expandable lines
import { OpenRouter, tool } from '@openrouter/agent';
import { z } from 'zod';

const weatherTool = tool({
  name: 'get_weather',
  description: 'Get the current weather for a location',
  inputSchema: z.object({
    location: z.string().describe('City name, e.g., "San Francisco, CA"'),
  }),
  outputSchema: z.object({
    temperature: z.number(),
    conditions: z.string(),
  }),
  execute: async (params) => {
    // params is typed as { location: string }
    const weather = await fetchWeather(params.location);
    return {
      temperature: weather.temp,
      conditions: weather.description,
    };
  },
});
```

```python title="Python" lines
from openrouter_agent import tool
from pydantic import BaseModel, Field


class WeatherInput(BaseModel):
    location: str = Field(description='City name, e.g., "San Francisco, CA"')

    temperature: float
    conditions: str

    weather = await fetch_weather(params.location)
    return WeatherOutput(
        temperature=weather.temp,
        conditions=weather.description,
    )


weather_tool = tool(
    name="get_weather",
    description="Get the current weather for a location",
    input_schema=WeatherInput,
    output_schema=WeatherOutput,
    execute=get_weather,
)

```

## Tool Types

The SDK supports four types of tools, automatically detected from your configuration:

### Regular Tools

Standard tools with an execute function:

<CodeGroup>

```typescript lines
const calculatorTool = tool({
  name: 'calculate',
  description: 'Perform a mathematical calculation',
  inputSchema: z.object({
    expression: z.string().describe('Math expression like "2 + 2"'),
  }),
  outputSchema: z.object({
    result: z.number(),
  }),
  execute: async (params) => {
    const result = eval(params.expression); // Use a safer eval in production
    return { result };
  },
});
```

```python title="Python" lines
import ast
import operator

from openrouter_agent import tool
from pydantic import BaseModel, Field


class CalculatorInput(BaseModel):
    expression: str = Field(description='Math expression like "2 + 2"')


class CalculatorOutput(BaseModel):
    result: float


BINARY_OPS = {
    ast.Add: operator.add,
    ast.Sub: operator.sub,
    ast.Mult: operator.mul,
    ast.Div: operator.truediv,
    ast.Mod: operator.mod,
}
UNARY_OPS = {ast.UAdd: operator.pos, ast.USub: operator.neg}


# The expression comes from the model, so evaluate arithmetic only and never call eval() on it
def evaluate_arithmetic(expression: str) -> float:
    def evaluate(node: ast.AST) -> float:
        if isinstance(node, ast.Expression):
            return evaluate(node.body)
        if isinstance(node, ast.Constant) and type(node.value) in (int, float):
            return node.value
        if isinstance(node, ast.BinOp) and type(node.op) in BINARY_OPS:
            return BINARY_OPS[type(node.op)](evaluate(node.left), evaluate(node.right))
        if isinstance(node, ast.UnaryOp) and type(node.op) in UNARY_OPS:
            return UNARY_OPS[type(node.op)](evaluate(node.operand))
        raise ValueError(f"Unsupported expression: {expression!r}")

    return evaluate(ast.parse(expression, mode="eval"))


async def calculate(params: CalculatorInput, ctx) -> CalculatorOutput:
    return CalculatorOutput(result=evaluate_arithmetic(params.expression))


calculator_tool = tool(
    name="calculate",
    description="Perform a mathematical calculation",
    input_schema=CalculatorInput,
    output_schema=CalculatorOutput,
    execute=calculate,
)

```

```go title="Go" lines
type CalculateParams struct {
	Expression string `json:"expression"`
}

type CalculateResult struct {
	Result float64 `json:"result"`
}

var calculatorTool = map[string]any{
	"type": "function",
	"function": map[string]any{
		"name":        "calculate",
		"description": "Perform a mathematical calculation",
		"parameters": map[string]any{
			"type": "object",
			"properties": map[string]any{
				"expression": map[string]any{"type": "string", "description": `Math expression like "2 + 2"`},
			},
			"required": []string{"expression"},
		},
	},
}

func executeCalculate(params CalculateParams) (CalculateResult, error) {
	// Evaluates constant Go expressions only; use a safer evaluator in production.
	// Go rules apply, so integer operands divide as integers: 5/2 is 2, 5.0/2 is 2.5.
	tv, err := types.Eval(token.NewFileSet(), nil, token.NoPos, params.Expression)
	if err != nil || tv.Value == nil {
		return CalculateResult{}, fmt.Errorf("invalid expression: %q", params.Expression)
	}
	// ToFloat yields an Unknown value for booleans and strings: reject them instead of returning 0
	value := constant.ToFloat(tv.Value)
	if value.Kind() != constant.Float {
		return CalculateResult{}, fmt.Errorf("not a numeric expression: %q", params.Expression)
	}
	result, _ := constant.Float64Val(value)
	return CalculateResult{Result: result}, nil
}
```

### Generator Tools

Tools that yield progress updates during execution. Add `eventSchema` to enable generator mode:

<CodeGroup>

```typescript expandable lines
const searchTool = tool({
  name: 'search_database',
  description: 'Search documents with progress updates',
  inputSchema: z.object({
    query: z.string(),
    limit: z.number().default(10),
  }),
  // eventSchema triggers generator mode
  eventSchema: z.object({
    progress: z.number().min(0).max(100),
    message: z.string(),
  }),
  outputSchema: z.object({
    results: z.array(z.string()),
    totalFound: z.number(),
  }),
  // execute is now an async generator
  execute: async function* (params) {
    yield { progress: 0, message: 'Starting search...' };

    const results = [];
    for (let i = 0; i < 5; i++) {
      yield { progress: (i + 1) * 20, message: `Searching batch ${i + 1}...` };
      results.push(...await searchBatch(params.query, i));
    }

    // Final yield is the output
    yield { progress: 100, message: 'Complete!' };

    // Return the final result (or yield it as last value)
    return {
      results: results.slice(0, params.limit),
      totalFound: results.length,
    };
  },
});
```

```python title="Python" lines
from openrouter_agent import tool
from pydantic import BaseModel, Field


class SearchInput(BaseModel):
    query: str
    limit: int = 10


# event_schema triggers generator mode
class SearchProgress(BaseModel):
    progress: int = Field(ge=0, le=100)
    message: str


class SearchOutput(BaseModel):
    results: list[str]
    total_found: int


# execute is now an async generator
async def search_database(params: SearchInput, ctx):
    yield SearchProgress(progress=0, message="Starting search...")

    results = []
    for i in range(5):
        yield SearchProgress(progress=(i + 1) * 20, message=f"Searching batch {i + 1}...")
        results.extend(await search_batch(params.query, i))

    yield SearchProgress(progress=100, message="Complete!")

    # The final yield is the output
    yield SearchOutput(
        results=results[: params.limit],
        total_found=len(results),
    )


search_tool = tool(
    name="search_database",
    description="Search documents with progress updates",
    input_schema=SearchInput,
    event_schema=SearchProgress,
    output_schema=SearchOutput,
    execute=search_database,
)

```

```go title="Go" lines
type SearchParams struct {
	Query string `json:"query"`
	Limit int    `json:"limit"`
}

// SearchProgress is the event type emitted while the tool runs
type SearchProgress struct {
	Progress int    `json:"progress"`
	Message  string `json:"message"`
}

type SearchResult struct {
	Results    []string `json:"results"`
	TotalFound int      `json:"totalFound"`
}

var searchTool = map[string]any{
	"type": "function",
	"function": map[string]any{
		"name":        "search_database",
		"description": "Search documents with progress updates",
		"parameters": map[string]any{
			"type": "object",
			"properties": map[string]any{
				"query": map[string]any{"type": "string"},
				"limit": map[string]any{"type": "integer", "default": 10},
			},
			"required": []string{"query"},
		},
	},
}

// emit receives progress updates; the return value is the final result
func executeSearch(params SearchParams, emit func(SearchProgress)) (SearchResult, error) {
	// The limit comes from the model, so treat a missing or non-positive value as the default
	if params.Limit <= 0 {
		params.Limit = 10
	}
	emit(SearchProgress{Progress: 0, Message: "Starting search..."})

	var results []string
	for i := 0; i < 5; i++ {
		emit(SearchProgress{Progress: (i + 1) * 20, Message: fmt.Sprintf("Searching batch %d...", i+1)})
		batch, err := searchBatch(params.Query, i)
		if err != nil {
			return SearchResult{}, err
		}
		results = append(results, batch...)
	}

	emit(SearchProgress{Progress: 100, Message: "Complete!"})

	return SearchResult{
		Results:    results[:min(params.Limit, len(results))],
		TotalFound: len(results),
	}, nil
}
```

</CodeGroup>

Use `getToolCalls()` to retrieve manual tool calls for processing.

When manual tool calls are left unresolved at the end of a round, the loop pauses and the calls are persisted to `ConversationState.pendingToolCalls` with `status: 'awaiting_client_tools'`, so cold-start consumers can recover them from serialized state. To continue, call `callModel` again with new `input`; the stale pendings are cleared and the run proceeds as a normal turn. Manual tools are not approved or rejected by call ID the way [approval](/agent-sdk/call-model/tool-approval-state) and [HITL](#human-in-the-loop-hitl-tools) flows are. See [Status Values](/agent-sdk/call-model/tool-approval-state#status-values).

### Human-in-the-Loop (HITL) Tools

HITL tools extend manual-tool semantics with two sync-or-async hooks that let you decide per call whether to respond programmatically or pause for a human:

- `onToolCalled` fires when the model invokes the tool. Return a value to feed the model directly (like a regular `execute`), or return `null` to pause the loop like a manual tool. The caller resumes later by supplying a `function_call_output` item.
- `onResponseReceived` is optional. It fires on a later turn when an incoming `function_call_output` matches a prior call of this tool (by `callId → function_call.name`). It receives the caller-supplied raw result and returns the value sent to the model. Throwing surfaces as a tool error to the model.

An `outputSchema` is required for HITL tools. It validates both the `onToolCalled` return value (when non-null) and the value delivered via `function_call_output` (whether transformed by `onResponseReceived` or passed through directly).

<CodeGroup>

```typescript expandable lines
const approvePaymentTool = tool({
  name: 'approve_payment',
  description: 'Approve a payment, escalating large amounts to a human',
  inputSchema: z.object({
    amount: z.number(),
    recipient: z.string(),
  }),
  outputSchema: z.object({
    ok: z.boolean(),
    reviewedAt: z.number().optional(),
  }),
  onToolCalled: async (input) => {
    // Auto-approve small amounts
    if (input.amount < 100) {
      return { ok: true };
    }
    // Escalate to a human — pauses the loop
    return null;
  },
  onResponseReceived: async (raw) => {
    // Post-process the caller-supplied result before the model sees it
    return { ...(raw as object), reviewedAt: Date.now() };
  },
});
```

```python title="Python" lines
import time

from openrouter_agent import tool
from pydantic import BaseModel


class PaymentInput(BaseModel):
    amount: float
    recipient: str


class PaymentDecision(BaseModel):
    ok: bool
    reviewed_at: int | None = None


async def on_tool_called(params: PaymentInput, ctx=None):
    # Auto-approve small amounts
    if params.amount < 100:
        return PaymentDecision(ok=True)
    # Escalate to a human — pauses the loop
    return None


async def on_response_received(raw, ctx=None):
    # Post-process the caller-supplied result before the model sees it
    return {**dict(raw), "reviewed_at": int(time.time() * 1000)}


approve_payment_tool = tool(
    name="approve_payment",
    description="Approve a payment, escalating large amounts to a human",
    input_schema=PaymentInput,
    output_schema=PaymentDecision,
    on_tool_called=on_tool_called,
    on_response_received=on_response_received,
)

```

```go title="Go" lines
type PaymentParams struct {
	Amount    float64 `json:"amount"`
	Recipient string  `json:"recipient"`
}

// PaymentResult is the output schema for both paths
type PaymentResult struct {
	OK         bool   `json:"ok"`
	ReviewedAt *int64 `json:"reviewedAt,omitempty"`
}

var approvePaymentTool = map[string]any{
	"type": "function",
	"function": map[string]any{
		"name":        "approve_payment",
		"description": "Approve a payment, escalating large amounts to a human",
		"parameters": map[string]any{
			"type": "object",
			"properties": map[string]any{
				"amount":    map[string]any{"type": "number"},
				"recipient": map[string]any{"type": "string"},
			},
			"required": []string{"amount", "recipient"},
		},
	},
}

// onToolCalled returns nil to escalate to a human, which pauses your loop
func onToolCalled(input PaymentParams) *PaymentResult {
	// Auto-approve small amounts
	if input.Amount < 100 {
		return &PaymentResult{OK: true}
	}
	return nil
}

// onResponseReceived post-processes the caller-supplied result before the model sees it
func onResponseReceived(raw PaymentResult) PaymentResult {
	now := time.Now().UnixMilli()
	raw.ReviewedAt = &now
	return raw
}
```

</CodeGroup>

When `onToolCalled` returns `null`, the conversation state moves to `status: 'awaiting_hitl'` and the paused call surfaces via `getToolCalls()` / `getPendingToolCalls()`. Resume by calling `callModel` again with a `function_call_output` item for each paused call in the input.

<Note>
HITL tools differ from `requireApproval`: approval gates pause *before* execution for a yes/no decision, while HITL tools let `onToolCalled` run arbitrary logic first and only pause when it returns `null`. Use HITL when the decision is data-driven (e.g., amount thresholds, risk scoring); use `requireApproval` when you always want explicit human consent. See [Tool Approval & State](/agent-sdk/call-model/tool-approval-state).
</Note>

## Schema Definition

### Input Schema

Define what parameters the tool accepts:

```typescript expandable lines
const inputSchema = z.object({
  // Required parameters
  query: z.string().describe('Search query'),

  // Optional with default
  limit: z.number().default(10).describe('Max results'),

  // Optional without default
  filter: z.string().optional().describe('Filter expression'),

  // Enum values
  sortBy: z.enum(['relevance', 'date', 'popularity']).default('relevance'),

  // Nested objects
  options: z.object({
    caseSensitive: z.boolean().default(false),
    wholeWord: z.boolean().default(false),
  }).optional(),

  // Arrays
  tags: z.array(z.string()).optional(),
});
```

### Output Schema

Define the structure of results returned to the model:

```typescript lines
const outputSchema = z.object({
  results: z.array(z.object({
    id: z.string(),
    title: z.string(),
    score: z.number(),
  })),
  metadata: z.object({
    totalCount: z.number(),
    searchTimeMs: z.number(),
  }),
});
```

### Event Schema (Generator Tools)

Define progress/status events for generator tools:

```typescript lines
const eventSchema = z.object({
  stage: z.enum(['initializing', 'processing', 'finalizing']),
  progress: z.number(),
  currentItem: z.string().optional(),
});
```

```python title="Python" lines

from pydantic import BaseModel

    conditions: str

    output_schema=WeatherOutput,

async def main():
    openrouter = OpenRouter(api_key=os.environ["OPENROUTER_API_KEY"])
```

```go title="Go" lines
package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	"io"
	"log"
	"net/http"
	"os"
)

type ToolCall struct {
	ID       string `json:"id"`
	Type     string `json:"type"`
	Function struct {
		Name      string `json:"name"`
		Arguments string `json:"arguments"`
	} `json:"function"`
}

type Message struct {
	Role       string     `json:"role"`
	Content    any        `json:"content"`
	ToolCalls  []ToolCall `json:"tool_calls,omitempty"`
	ToolCallID string     `json:"tool_call_id,omitempty"`
}

var weatherTool = map[string]any{
	"type": "function",
	"function": map[string]any{
		"name":        "get_weather",
		"description": "Get current weather for a location",
		"parameters": map[string]any{
			"type":       "object",
			"properties": map[string]any{"location": map[string]any{"type": "string"}},
			"required":   []string{"location"},
		},
	},
}

func getWeather(args string) string {
	var in struct {
		Location string `json:"location"`
	}
	json.Unmarshal([]byte(args), &in)
	out, _ := json.Marshal(map[string]any{"location": in.Location, "temperature": 22, "conditions": "sunny"})
	return string(out)
}

func main() {
	messages := []Message{{Role: "user", Content: "What is the weather in Tokyo?"}}

	// Tools are executed by this loop until the model answers in text
	for i := 0; i < 5; i++ {
		body, _ := json.Marshal(map[string]any{
			"model":    "openai/gpt-5-nano",
			"messages": messages,
			"tools":    []any{weatherTool},
		})
		req, _ := http.NewRequest("POST", "https://openrouter.ai/api/v1/chat/completions", bytes.NewReader(body))
		req.Header.Set("Authorization", "Bearer "+os.Getenv("OPENROUTER_API_KEY"))
		req.Header.Set("Content-Type", "application/json")
		resp, err := http.DefaultClient.Do(req)
		if err != nil {
			log.Fatal(err)
		}
		raw, _ := io.ReadAll(resp.Body)
		resp.Body.Close()

		var res struct {
			Choices []struct {
				Message Message `json:"message"`
			} `json:"choices"`
		}
		if err := json.Unmarshal(raw, &res); err != nil || len(res.Choices) == 0 {
			log.Fatalf("unexpected response (%d): %s", resp.StatusCode, raw)
		}
		msg := res.Choices[0].Message
		messages = append(messages, msg)

		if len(msg.ToolCalls) == 0 {
			fmt.Println(msg.Content)
			return
		}
		for _, tc := range msg.ToolCalls {
			messages = append(messages, Message{Role: "tool", ToolCallID: tc.ID, Content: getWeather(tc.Function.Arguments)})
		}
	}
}
```

### Type-Safe Tool Calls with `as const`

Use `as const` for full type inference on tool calls:

<CodeGroup>

```typescript lines
const result = callModel(openrouter, {
  model: 'openai/gpt-5-nano',
  input: 'What is the weather?',
  tools: [weatherTool, searchTool] as const,
});

// Tool calls are typed as union of tool inputs
for await (const toolCall of result.getToolCallsStream()) {
  if (toolCall.name === 'get_weather') {
    // toolCall.arguments is typed as { location: string }
    console.log('Weather for:', toolCall.arguments.location);
  }
}
```

```python title="Python" lines
import asyncio
import os

from pydantic import BaseModel
from openrouter_agent import OpenRouter, call_model, tool


class WeatherInput(BaseModel):
    location: str


class WeatherOutput(BaseModel):
    temperature: float
    conditions: str


class SearchInput(BaseModel):
    query: str


class SearchOutput(BaseModel):
    results: list[str]


weather_tool = tool(
    name="get_weather",
    description="Get current weather for a location",
    input_schema=WeatherInput,
    output_schema=WeatherOutput,
    execute=lambda params, ctx: WeatherOutput(temperature=22, conditions="sunny"),
)

search_tool = tool(
    name="search",
    description="Search the web",
    input_schema=SearchInput,
    output_schema=SearchOutput,
    execute=lambda params, ctx: SearchOutput(results=[f"Result for {params.query}"]),
)


async def main():
    openrouter = OpenRouter(api_key=os.environ["OPENROUTER_API_KEY"])

    result = call_model(
        openrouter,
        {
            "model": "openai/gpt-5-nano",
            "input": "What is the weather in San Francisco?",
            "tools": [weather_tool, search_tool],
        },
    )

    # Tool calls are validated against each tool's input schema
    async for tool_call in result.get_tool_calls_stream():
        if tool_call.name == "get_weather":
            # tool_call.arguments matches WeatherInput: {"location": str}
            print("Weather for:", tool_call.arguments["location"])

    print("Final text:", await result.get_text())


asyncio.run(main())

```

## Execute Context

Tool execute functions receive a flat context object as their second argument. It merges `TurnContext` fields with a `tools` map and a `setContext()` method:

<CodeGroup>

```typescript lines
const contextAwareTool = tool({
  name: 'context_tool',
  inputSchema: z.object({ data: z.string() }),
  outputSchema: z.object({ result: z.string() }),
  execute: async (params, context) => {
    // TurnContext fields are available directly
    console.log('Turn:', context.numberOfTurns);
    console.log('History:', context.turnRequest?.input);
    console.log('Model:', context.turnRequest?.model);

    return {
      result: `Processed on turn ${context.numberOfTurns}`,
    };
  },
});
```

```python title="Python" lines
class ContextInput(BaseModel):
    data: str


class ContextOutput(BaseModel):
    result: str


async def execute_context_tool(params, ctx):
    # TurnContext fields are available directly
    turn_request = ctx.get("turn_request")
    print("Turn:", ctx.get("number_of_turns"))
    print("History:", getattr(turn_request, "input", None))
    print("Model:", getattr(turn_request, "model", None))

    return ContextOutput(result=f"Processed on turn {ctx.get('number_of_turns')}")


context_aware_tool = tool(
    name="context_tool",
    input_schema=ContextInput,
    output_schema=ContextOutput,
    execute=execute_context_tool,
)
```

```go title="Go" lines

	Name: "context_tool",
	Parameters: map[string]any{

		"properties": map[string]any{"data": map[string]any{"type": "string"}},
		"required":   []string{"data"},

		}
	},
}
```

</CodeGroup>

### Context Properties

| Property | Type | Description |
|----------|------|-------------|
| `numberOfTurns` | `number` | Current turn number (1-indexed) |
| `turnRequest` | `OpenResponsesRequest \| undefined` | Current request object |
| `toolCall` | `OpenResponsesFunctionToolCall \| undefined` | The tool call being executed |
| `local` | `Readonly<TContext>` | This tool's own context (read-only) |
| `setContext` | `(partial: Partial<TContext>) => void` | Mutate this tool's context |
| `shared` | `Readonly<TShared>` | Shared context visible to all tools |
| `setSharedContext` | `(partial: Partial<TShared>) => void` | Mutate shared context |

## Tool Context

Tools can declare a `contextSchema` to receive typed, persistent context data from the caller. Context is keyed by tool name and persists across turns.

### Declaring contextSchema

<CodeGroup>

```typescript expandable lines
const weatherTool = tool({
  name: 'get_weather',
  description: 'Get weather for a location',
  inputSchema: z.object({
    location: z.string(),
  }),
  outputSchema: z.object({
    temperature: z.number(),
  }),
  // Declare what context this tool needs
  contextSchema: z.object({
    apiKey: z.string(),
    units: z.enum(['celsius', 'fahrenheit']),
  }),
  execute: async (params, context) => {
    // Access this tool's own context via local
    const { apiKey, units } = context.local;

    const weather = await fetchWeather(
      params.location,
      apiKey,
      units,
    );
    return { temperature: weather.temp };
  },
});
```

### Mutating Context with setContext

Tools can update their own context using `setContext()`. Changes persist across turns via the shared store and are visible immediately. `context.local` is a live getter that always reads the latest values:

<CodeGroup>

```typescript expandable lines
const authTool = tool({
  name: 'auth',
  inputSchema: z.object({ action: z.string() }),
  contextSchema: z.object({
    token: z.string(),
    refreshCount: z.number(),
  }),
  execute: async (params, context) => {
    const { token } = context.local;

    if (isExpired(token)) {
      const newToken = await refreshToken(token);
      // Mutate own context — persists to next turn
      context.setContext({
        token: newToken,
        refreshCount:
          context.local.refreshCount + 1,
      });
    }

    return { success: true };
  },
});
```

### Shared Context

Use `sharedSchema` on `tool()` and `sharedContextSchema` on `callModel` to share typed state across tools:

<CodeGroup>

```typescript expandable lines
const SharedContextSchema = z.object({
  _sessionId: z.string().optional(),
});

const execTool = tool({
  name: 'sandbox_exec',
  inputSchema: z.object({ command: z.string() }),
  sharedSchema: SharedContextSchema,
  execute: async (input, ctx) => {
    // Read shared state set by any tool
    const sid = ctx.shared._sessionId;
    const session = await connect(sid);
    // Write shared state for other tools
    ctx.setSharedContext({ _sessionId: session.id });
    return await session.exec(input.command);
  },
});

const result = openrouter.callModel({
  model: 'openai/gpt-5-nano',
  input: 'Run a command',
  tools: [execTool] as const,
  sharedContextSchema: SharedContextSchema,
  context: {
    shared: { _sessionId: 'existing-session' },
    sandbox_exec: {},
  },
});
```

```python title="Python" lines
class SharedContext(BaseModel):
    session_id: str | None = None


class ExecInput(BaseModel):
    command: str


async def execute_sandbox(params, ctx):
    # Read shared state set by any tool (validated against SharedContext)
    shared = SharedContext.model_validate(ctx["shared"] or {})
    session = await connect(shared.session_id)
    # Write shared state for other tools
    ctx["set_shared_context"]({"session_id": session.id})
    return await session.exec(params.command)


exec_tool = tool(
    name="sandbox_exec",
    description="Run a shell command in the sandbox",
    input_schema=ExecInput,
    execute=execute_sandbox,
)

result = call_model(
    client,
    {
        "model": "openai/gpt-5-nano",
        "input": "Use the sandbox_exec tool to run the command `echo hello`.",
        "tools": [exec_tool],
        "context": {
            "shared": {"session_id": "existing-session"},
            "sandbox_exec": {},
        },
    },
)

```

```go title="Go" lines
sharedContextSchema := map[string]any{
	"type": "object",
	"properties": map[string]any{
		"_sessionId": map[string]any{"type": "string"},
	},
}

execTool := Tool{
	Name:        "sandbox_exec",
	Description: "Run a shell command in the sandbox session",
	Parameters: map[string]any{
		"type":       "object",
		"properties": map[string]any{"command": map[string]any{"type": "string"}},
		"required":   []string{"command"},
	},
	SharedSchema: sharedContextSchema,
	Execute: func(input map[string]any, ctx TurnContext) map[string]any {
		// Read shared state set by any tool
		sid, _ := ctx.Shared["_sessionId"].(string)
		session := connect(sid)
		// Write shared state for other tools
		ctx.SetSharedContext(map[string]any{"_sessionId": session.ID})
		command, _ := input["command"].(string)
		return session.Exec(command)
	},
}

result := callModel(CallModelRequest{
	Model:               "openai/gpt-5-nano",
	Input:               "Run the command `echo hello` in the sandbox",
	Tools:               []Tool{execTool},
	SharedContextSchema: sharedContextSchema,
	Context: map[string]map[string]any{
		"shared":       {"_sessionId": "existing-session"},
		"sandbox_exec": {},
	},
})
```

</CodeGroup>

<Note>
`context.local` is scoped to one tool. `context.shared` is visible to all tools and persists across turns. Pass the same `sharedSchema` to each tool for typed access, and `sharedContextSchema` to `callModel` for runtime validation.
</Note>

## Tool Execution

callModel automatically executes tools and handles multi-turn conversations. When the model calls a tool, the SDK executes it, sends the result back, and continues until the model provides a final response.

### Automatic Execution Flow

When you provide tools with execute functions:

The Python tab starts with the setup that the other Python samples on this page assume: the imports, plus a `client`. The Go tabs on this page are fragments, because there is no Go agent SDK. `Tool`, `TurnContext`, `CallModelRequest`, `Step`, and `callModel` in the Go tabs of this section stand for a tool loop that your application defines; they are not imported from a library. Later Go tabs call the chat completions endpoint (`/api/v1/chat/completions`) directly through `chat` and `complete`, which are also application-defined helpers: each one sends the messages and tool definitions and exits on an HTTP error or an empty `choices` list. `chat` returns the first choice's message and `complete` returns the whole response.

<CodeGroup>

```typescript expandable lines
import { OpenRouter, tool } from '@openrouter/agent';
import { z } from 'zod';

const weatherTool = tool({
  name: 'get_weather',
  inputSchema: z.object({ location: z.string() }),
  outputSchema: z.object({ temperature: z.number() }),
  execute: async ({ location }) => {
    return { temperature: await fetchTemperature(location) };
  },
});

const result = openrouter.callModel({
  model: 'openai/gpt-5-nano',
  input: 'What is the weather in Paris?',
  tools: [weatherTool],
});

// getText() waits for all tool execution to complete
const text = await result.getText();
// "The weather in Paris is 18°C."
```

### Execution Sequence

1. Model receives prompt and generates tool call
2. SDK extracts tool call and validates arguments
3. Tool's execute function runs
4. Result is formatted and sent back to model
5. Model generates final response (or more tool calls)
6. Process repeats until model is done

### Controlling Execution Rounds

#### maxToolRounds (Number)

Limit the maximum number of tool execution rounds:

<CodeGroup>

```typescript lines
const result = openrouter.callModel({
  model: 'openai/gpt-5-nano',
  input: 'Research this topic thoroughly',
  tools: [searchTool, analyzeTool],
  maxToolRounds: 3, // Stop after 3 rounds of tool execution
});
```

```python title="Python" lines
from openrouter_agent import step_count_is

result = call_model(
    client,
    {
        "model": "openai/gpt-5-nano",
        "input": "Research this topic thoroughly",
        "tools": [search_tool, analyze_tool],
        "stop_when": step_count_is(3),  # Stop after 3 rounds of tool execution
    },

```

```go title="Go" lines
result := callModel(CallModelRequest{
	Model:         "openai/gpt-5-nano",
	Input:         "Research this topic thoroughly",
	Tools:         []Tool{searchTool, analyzeTool},
```

</CodeGroup>

Setting `maxToolRounds: 0` disables automatic execution - you get raw tool calls.

### Accessing Tool Calls

#### getToolCalls()

Get the tool calls from the first response. `stepCountIs(1)` ends the run after one tool round, but a tool with an `execute` function still runs before the run ends. To read the calls without running anything, define the tool with `execute: false` as shown in [Manual Tools](#manual-tools):

<CodeGroup>

```typescript lines
const result = openrouter.callModel({
  model: 'openai/gpt-5-nano',
  input: 'What is the weather in Tokyo and Paris?',
  tools: [weatherTool] as const,
  stopWhen: stepCountIs(1), // Stop after the first tool round
  allowFinalResponse: false, // No extra final-answer turn
});

const toolCalls = await result.getToolCalls();

for (const call of toolCalls) {
  console.log(`Tool: ${call.name}`);
  console.log(`ID: ${call.id}`);
  console.log(`Arguments:`, call.arguments);
}
```

```python title="Python" lines
from openrouter_agent import step_count_is

async def main():
```

#### ToolResultEvent Type

```typescript lines
type ToolResultEvent<TResult = unknown, TPreliminaryResults = unknown> = {
  type: 'tool.result';
  toolCallId: string;
  /**
   * Origin of the tool: `'client'` for locally-defined tools (result is
   * precisely typed from your `outputSchema`), `'mcp'` for tools wrapped from a
   * remote MCP server (result is `unknown`).
   */
  source: 'client' | 'mcp';
  result: TResult;
  timestamp: number;
  preliminaryResults?: TPreliminaryResults[];
};
```

The `tool.result` event provides the final output from tool execution along with all intermediate `preliminaryResults` that were yielded during execution (for generator tools). This is useful when you need both real-time progress updates and a summary of all progress at completion.

<Warning>
The `source` field was added in `@openrouter/agent` 0.8.0. This is an additive breaking change for consumers that exhaustively matched (or constructed) `tool.result` events; add a `source` case, or spread it through, to keep type-checking clean.
</Warning>

#### Narrowing MCP vs. client results

Mixing an MCP tool (whose output schema is `unknown`) with fully-typed local tools used to collapse the entire result union to `unknown`. The `source` discriminant fixes that: narrow on `source === 'client'` to recover your schema-derived result types, and treat `source === 'mcp'` results as `unknown`.

<CodeGroup>

```typescript lines
for await (const event of result.getFullResponsesStream()) {
  if (event.type !== 'tool.result') continue;

  if (event.source === 'client') {
    // event.result is typed from the tool's outputSchema union
    console.log('Local tool result:', event.result);
  } else {
    // event.source === 'mcp': result stays unknown; validate before use
    console.log('MCP tool result (unknown):', event.result);
  }
}
```

```typescript lines
      const response = await fetch(params.url);
      if (!response.ok) {
        return { error: `HTTP ${response.status}: ${response.statusText}` };
      }
      return { data: await response.json() };
    } catch (error) {
      return { error: `Failed to fetch: ${error.message}` };
    }
  },
});
```

```python title="Python" lines
from typing import Any

import httpx
from pydantic import BaseModel, HttpUrl

class FetchInput(BaseModel):
    url: HttpUrl


class FetchOutput(BaseModel):
    data: Any | None = None
    error: str | None = None


# The URL comes from the model, so only fetch from the hosts this tool is meant to reach
ALLOWED_HOSTS = {"httpbin.org"}


async def fetch_data(params, ctx):
    if params.url.scheme != "https" or params.url.host not in ALLOWED_HOSTS:
        return FetchOutput(error=f"Host not allowed: {params.url.host}")
    try:
        # Redirects stay off: a redirect could lead to a host outside the allow-list
        async with httpx.AsyncClient(timeout=10, follow_redirects=False) as http:
            response = await http.get(str(params.url))
        if not response.is_success:
            return FetchOutput(error=f"HTTP {response.status_code}: {response.reason_phrase}")
        return FetchOutput(data=response.json())
    except Exception as error:
        return FetchOutput(error=f"Failed to fetch: {error}")


robust_tool = tool(
    name="fetch_data",
    input_schema=FetchInput,
    output_schema=FetchOutput,
    execute=fetch_data,
)
```

### Idempotent Tools

Design tools to be safely re-executable. A lookup followed by a separate create is not atomic: two concurrent calls with the same email can both pass the lookup. These samples rely on the backing store enforcing a unique constraint on email, so the second create fails instead of adding a duplicate. If your store has no such constraint, replace the two steps with a single atomic create-or-get (upsert).

<CodeGroup>

```typescript lines
const createUserTool = tool({
  name: 'create_user',
  inputSchema: z.object({
    email: z.string().email(),
    name: z.string(),
  }),
  execute: async (params) => {
    // Check if user exists first
    const existing = await findUserByEmail(params.email);
    if (existing) {
      return { userId: existing.id, created: false };
    }

    const user = await createUser(params);
    return { userId: user.id, created: true };
  },
});
```

### Timeout Handling

Wrap long-running operations:

<CodeGroup>

```typescript lines
const longRunningTool = tool({
  name: 'process_data',
  inputSchema: z.object({ dataId: z.string() }),
  execute: async (params) => {
    const timeoutMs = 30000;

    const result = await Promise.race([
      processData(params.dataId),
      new Promise((_, reject) =>
        setTimeout(() => reject(new Error('Operation timed out')), timeoutMs)
      ),
    ]);

    return result;
  },
});
```

```python title="Python" lines
class ProcessDataInput(BaseModel):
    data_id: str


async def process_data_execute(params, ctx):
    timeout_s = 30

    try:
        return await asyncio.wait_for(process_data(params.data_id), timeout=timeout_s)
    except asyncio.TimeoutError:
        raise Exception("Operation timed out")


long_running_tool = tool(
    name="process_data",
    input_schema=ProcessDataInput,
    execute=process_data_execute,
)

```

```go title="Go" lines
longRunningTool := map[string]any{

		"parameters": map[string]any{
			"type":       "object",

			"required":   []string{"dataId"},
		},
	},

longRunningExec := func(args string) (string, error) {
	var params struct {
		DataID string `json:"dataId"`
	}
	if err := json.Unmarshal([]byte(args), &params); err != nil {
		return "", err

	ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
	defer cancel()

	// processData stops its work when ctx is cancelled, so nothing keeps
	// running after the timeout
	result, err := processData(ctx, params.DataID)
	if errors.Is(err, context.DeadlineExceeded) {
		return "", errors.New("operation timed out")
	}
	return result, err
}
```
